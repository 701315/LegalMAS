# src/workers/extract_worker.py
"""
Extract Worker — извлечение текста и таблиц из PDF, сохранённых в MinIO.

Позиция в пайплайне:
- Пайплайн: import → extract → pii → structure → classify → run_deal_checks.
- На вход получает документ (PDF) и создаёт артефакт text_layout_raw.

Что делает воркер:
- Читает исходный PDF из MinIO (по ссылке original_ref или converted_pdf_ref из DocumentVersion).
- Определяет стратегию извлечения (native text / OCR / гибрид).
- Строит структуру text_layout_raw (элементы и таблицы) и сохраняет её в MinIO.
- При низком качестве извлечения или ошибках отправляет документ в карантин.
- При успехе обновляет статус документа до EXTRACTED и запускает PII‑санитизацию.

Архитектура:
- Внешняя Celery‑таска extract_document — синхронная (def), её вызывает Celery.
- Внутри таски один раз происходит переход sync → async через run_async.
- Вся логика работы с БД и MinIO сосредоточена в _extract_document_async()
  и вспомогательных async‑функциях под управлением db_uow(), без ручного
  управления event loop.
- Метрики производительности и качества извлечения пишутся в Prometheus
  через глобальный реестр METRICS.
"""

import time
import uuid

from sqlalchemy.ext.asyncio import AsyncSession

from src.ops.metrics import METRICS
from core.config import settings
from core.constants import DocumentStatus
from core.exceptions import PermanentError, QuarantineError
from core.logging import get_logger
from db.repositories import DocumentRepository, DocumentVersionRepository
from db.uow import db_uow, UnitOfWork
from mas_queue.base_task import MASBaseTask
from mas_queue.celery_app import celery_app
from mas_queue.async_helpers import run_async
from services.pdf_extractor import (
    ExtractionResult,
    build_text_layout_raw,
    extract_pdf,
)
from services.layout_language_filter import (
    detect_languages_in_layout,
    filter_layout_to_language,
)
from services.deal_language_service import update_deal_languages
from db.repositories.deal_repo import DealRepository

from storage.artifact_store import ArtifactPath, ArtifactStore

logger = get_logger(__name__)


@celery_app.task(
    base=MASBaseTask,
    bind=True,
    name="workers.extract_document",
    track_job=False,
    queue="default",
)
def extract_document(
    self,
    document_version_id: str,
    document_id: str,
    deal_id: str,
    tenant_id: str,
    run_id: str | None = None,
) -> dict:
    """
    Sync Celery entrypoint.

    Новый архитектурный паттерн:
    - Celery вызывает синхронную функцию.
    - Внутри мы один раз передаём управление в async‑мир через run_async.
    - Весь async‑код (БД, MinIO и т.п.) живёт в _extract_document_async()
      и вспомогательных async‑функциях под db_uow().
    """

    async def _wrapper() -> dict:
        # Основной unit of work для извлечения
        async with db_uow() as uow:
            try:
                return await _extract_document_async(
                    document_version_id=document_version_id,
                    document_id=document_id,
                    deal_id=deal_id,
                    tenant_id=tenant_id,
                    db=uow.session,
                    run_id=run_id,
                )
            except PermanentError as e:
                logger.error(
                    "Permanent error during extraction",
                    document_id=document_id,
                    version_id=document_version_id,
                    error=str(e),
                )
                # Отдельный UoW только для записи статуса FAILED + деталей
                async with db_uow() as uow_status:
                    await _mark_failed(
                        uow_status,
                        document_uuid=uuid.UUID(document_id),
                        message=str(e),
                        run_id=uuid.UUID(run_id) if run_id else None,
                    )
                return {
                    "status": DocumentStatus.FAILED.value,
                    "document_id": document_id,
                    "error": str(e),
                }
            except QuarantineError as e:
                logger.error(
                    "Quarantine error during extraction",
                    document_id=document_id,
                    version_id=document_version_id,
                    error=str(e),
                )
                # Отдельный UoW только для записи статуса QUARANTINE + деталей
                async with db_uow() as uow_status:
                    await _mark_quarantine(
                        uow_status,
                        document_uuid=uuid.UUID(document_id),
                        error=e,
                        run_id=uuid.UUID(run_id) if run_id else None,
                    )
                return {
                    "status": DocumentStatus.QUARANTINE.value,
                    "document_id": document_id,
                    "error": str(e),
                }

    # Запускаем async‑обёртку через общий мост run_async
    return run_async(_wrapper())


async def _extract_document_async(
    document_version_id: str,
    document_id: str,
    deal_id: str,
    tenant_id: str,
    db: AsyncSession,
    run_id: str | None = None,
) -> dict:
    """
    Extract text and tables from PDF and produce text_layout_raw artifact.

    Все операции с БД выполняются через переданную AsyncSession (db),
    чтобы одна задача worker'а работала в рамках одного unit of work.
    """
    logger.info(
        "Starting extraction",
        document_id=document_id,
        version_id=document_version_id,
    )
    start_time = time.time()
    store = ArtifactStore(settings)

    # --- Чтение PDF из MinIO с защитой и логами ---
    try:
        pdf_ref = await get_pdf_source_ref(document_version_id, db)
        logger.info(
            "Extract: pdf_ref resolved",
            document_id=document_id,
            version_id=document_version_id,
            pdf_ref=pdf_ref,
        )

        if not pdf_ref:
            raise PermanentError(
                f"No PDF source ref for version {document_version_id} "
                f"(neither converted_pdf_ref nor original_ref is set)"
            )

        s3_path = pdf_ref.replace(f"s3://{store.bucket}/", "")
        logger.info(
            "Extract: reading PDF from store",
            document_id=document_id,
            version_id=document_version_id,
            s3_path=s3_path,
        )

        pdf_bytes = store.get_file(s3_path)
        logger.info(
            "Extract: PDF bytes loaded",
            document_id=document_id,
            version_id=document_version_id,
            size=len(pdf_bytes),
        )
    except Exception as e:
        logger.exception(
            "Extract: failed to read PDF from store",
            document_id=document_id,
            version_id=document_version_id,
            error=str(e),
        )
        raise PermanentError(f"Failed to read PDF from store: {e}")
    # --- КОНЕЦ БЛОКА ЧТЕНИЯ PDF ---

    # 2. Extract (Stage 4 PDF extractor: text / OCR / hybrid + tables)
    try:
        extraction: ExtractionResult = extract_pdf(pdf_bytes, document_id)
        logger.info(
            "Extract: pdf parsed",
            document_id=document_id,
            version_id=document_version_id,
            elements=len(extraction.elements),
            tables=len(extraction.tables),
            pages=extraction.page_count,
            strategy=extraction.strategy,
        )
    except Exception as e:
        logger.exception(
            "Extract: extract_pdf failed",
            document_id=document_id,
            version_id=document_version_id,
            error=str(e),
        )
        raise PermanentError(f"extract_pdf failed: {e}")

    # 3. Quarantine check
    _check_quarantine(extraction, document_id)

    # 4. Build text_layout_raw JSON (clause_uid = null at this stage)
    text_layout_raw = build_text_layout_raw(
        extraction,
        document_id=document_id,
        document_version_id=document_version_id,
    )

    # 4.1. Определяем фактические языки по исходному layout (multi-lang).
    detected_langs = detect_languages_in_layout(text_layout_raw)
    await update_deal_languages(
        db=db,
        deal_id=uuid.UUID(deal_id),
        new_languages=detected_langs,
    )

    # 4.2. Выбираем целевой язык для layout_primary.
    deal_repo = DealRepository(db)
    deal = await deal_repo.get_by_id(uuid.UUID(deal_id))

    primary = (getattr(deal, "primary_language", None) or "").lower()
    langs_from_deal = [
        (l or "").lower()
        for l in (getattr(deal, "languages", None) or [])
        if (l or "").strip()
    ]

    target_lang: str | None = None

    # 1) Явный primary_language сделки, если он реально найден в документе
    if primary and primary in detected_langs:
        target_lang = primary

    # 2) Первый язык сделки, который реально найден в документе
    if not target_lang:
        for lang in langs_from_deal:
            if lang in detected_langs:
                target_lang = lang
                break

    # 3) Доминирующий язык документа
    if not target_lang and detected_langs:
        target_lang = detected_langs[0]

    # 4) Явный fallback контракта пайплайна
    if not target_lang:
        target_lang = primary or (langs_from_deal[0] if langs_from_deal else "ru")

    # 4.3. Языковая фильтрация layout.
    #
    # is_bilingual определяем здесь, а не внутри filter_layout_to_language,
    # чтобы у extract_worker была та же картина, что у фильтра.
    is_bilingual = len(detected_langs) >= 2

    filter_context = {
        "deal_id": deal_id,
        "tenant_id": tenant_id,
        "document_id": document_id,
        "is_bilingual": is_bilingual,
    }
    layout_primary = filter_layout_to_language(
        text_layout_raw, target_lang, context=filter_context
    )

    # 4.4. Оценка качества одноязычного layout относительно исходного.
    #
    # Передаём is_bilingual чтобы пороги degraded учитывали тип документа:
    # - для билингвального документа ~50% элементов — это норма (один язык из двух);
    # - для монолингвального 50% элементов — это уже деградация.
    quality = _evaluate_primary_layout_quality(
        text_layout_raw, layout_primary, is_bilingual=is_bilingual
    )

    if not (layout_primary.get("elements") or []):
        logger.warning(
            "Primary layout is empty after language filter, falling back to raw layout",
            document_id=document_id,
            version_id=document_version_id,
            target_lang=target_lang,
        )
        layout_primary = text_layout_raw
    else:
        # Записываем диагностику внутрь metadata layout_primary
        metadata = dict(layout_primary.get("metadata") or {})
        lf_meta = dict(metadata.get("language_filter") or {})
        lf_meta.update(
            {
                "target_language": target_lang,
                "elements_before": quality["elements_before"],
                "elements_after": quality["elements_after"],
                "chars_before": quality["chars_before"],
                "chars_after": quality["chars_after"],
                "element_retention_ratio": quality["element_retention_ratio"],
                "char_retention_ratio": quality["char_retention_ratio"],
                "degraded": quality["degraded"],
                "degraded_reason": quality["degraded_reason"],
            }
        )
        metadata["language_filter"] = lf_meta
        layout_primary["metadata"] = metadata

        if quality["degraded"]:
            logger.warning(
                "Primary layout marked as degraded after language filter",
                document_id=document_id,
                version_id=document_version_id,
                target_lang=target_lang,
                element_retention_ratio=quality["element_retention_ratio"],
                char_retention_ratio=quality["char_retention_ratio"],
                degraded_reason=quality["degraded_reason"],
                is_bilingual=is_bilingual,
            )

    # 5. Upload to MinIO

    # 5.1. Полный многоязычный layout_raw (для отладки и fallback)
    raw_path = ArtifactPath.layout_raw(tenant_id, deal_id, document_id)
    raw_ref = store.put_json(raw_path, text_layout_raw)

    # 5.2. Одноязычный layout_primary (под выбранный язык)
    primary_path = ArtifactPath.layout_primary(tenant_id, deal_id, document_id)
    primary_ref = store.put_json(primary_path, layout_primary)

    # 6. Update DB (используем ту же сессию db)
    await update_extraction_result(
        document_version_id=document_version_id,
        document_id=document_id,
        raw_ref=raw_ref,
        primary_ref=primary_ref,
        page_count=extraction.page_count,
        strategy=extraction.strategy,
        ocr_confidence=extraction.avg_ocr_confidence,
        db=db,
    )

    duration = time.time() - start_time
    logger.info(
        "Extraction complete",
        document_id=document_id,
        elements=len(extraction.elements),
        tables=len(extraction.tables),
        pages=extraction.page_count,
        strategy=extraction.strategy,
        ocr_confidence=round(extraction.avg_ocr_confidence, 3),
        duration_seconds=round(duration, 2),
        target_lang=target_lang,
        is_bilingual=is_bilingual,
        element_retention_ratio=round(quality["element_retention_ratio"], 3),
    )

    # --- Метрики извлечения ---
    METRICS.extraction_duration.labels(
        strategy=extraction.strategy,
    ).observe(duration)

    METRICS.extraction_pages.labels(
        strategy=extraction.strategy,
    ).inc(extraction.page_count)

    if extraction.strategy == "ocr" and extraction.avg_ocr_confidence is not None:
        METRICS.extraction_ocr_confidence.observe(extraction.avg_ocr_confidence)

    # 7. Trigger PII Sanitizer (Stage 5)
    if not settings.DISABLE_DOWNSTREAM_TASKS:
        from mas_queue.celery_app import celery_app as _celery  # на случай циклов импорта

        _celery.send_task(
            "pii.sanitize_document",
            kwargs={
                "document_version_id": document_version_id,
            },
        )

    return {
        "document_id": document_id,
        "elements": len(extraction.elements),
        "tables": len(extraction.tables),
        "pages": extraction.page_count,
        "strategy": extraction.strategy,
        "duration_seconds": round(duration, 2),
    }


def _check_quarantine(extraction: ExtractionResult, document_id: str) -> None:
    """Raise QuarantineError if extraction results are below quality thresholds."""
    if extraction.errors and not extraction.elements:
        METRICS.quarantine.labels(reason="extraction_failed").inc()
        raise QuarantineError(
            f"PDF extraction failed: {'; '.join(extraction.errors)}",
            document_id=document_id,
            reason="extraction_failed",
        )

    if not extraction.elements:
        METRICS.quarantine.labels(reason="no_text_extracted").inc()
        raise QuarantineError(
            "No text extracted from PDF — 0 elements",
            document_id=document_id,
            reason="no_text_extracted",
        )

    if extraction.strategy in ("ocr", "hybrid") and (
        extraction.avg_ocr_confidence
        < settings.EXTRACTION_OCR_QUARANTINE_THRESHOLD
    ):
        METRICS.quarantine.labels(reason="ocr_low_confidence").inc()
        raise QuarantineError(
            (
                f"OCR confidence too low: "
                f"{extraction.avg_ocr_confidence:.2f} < "
                f"threshold {settings.EXTRACTION_OCR_QUARANTINE_THRESHOLD}"
            ),
            document_id=document_id,
            reason="ocr_low_confidence",
            details={
                "avg_confidence": extraction.avg_ocr_confidence,
                "threshold": settings.EXTRACTION_OCR_QUARANTINE_THRESHOLD,
            },
        )


def _evaluate_primary_layout_quality(
    raw_layout: dict,
    primary_layout: dict,
    is_bilingual: bool = False,
) -> dict:
    """
    Оценить качество layout_primary относительно исходного text_layout_raw.

    Возвращает словарь с полями:
    - elements_before / elements_after
    - chars_before / chars_after
    - element_retention_ratio
    - char_retention_ratio
    - degraded (bool)
    - degraded_reason (str | None)

    Пороги degraded зависят от типа документа:

    Монолингвальный документ:
      - Нормальный фильтр оставляет >80% элементов (убирает только шум).
      - Порог degraded: элементов < 60%, символов < 65%.

    Билингвальный документ (is_bilingual=True):
      - Нормальный фильтр оставляет ~50% элементов (один язык из двух).
      - Пороги должны быть ниже, иначе все двуязычные документы будут degraded.
      - Порог degraded: элементов < 20%, символов < 25%.
      - Почему 20%: если осталось меньше 1/5 от исходного — что-то точно сломалось
        (например, неверно определён целевой язык или фильтр упал в fallback
        и оставил только нейтральные блоки).
    """
    raw_elements = raw_layout.get("elements") or []
    prim_elements = primary_layout.get("elements") or []

    elements_before = len(raw_elements)
    elements_after = len(prim_elements)

    def _count_chars(text: str) -> int:
        return len((text or "").replace(" ", "").replace("\n", ""))

    chars_before = sum(_count_chars(el.get("text") or "") for el in raw_elements)
    chars_after = sum(_count_chars(el.get("text") or "") for el in prim_elements)

    elem_ratio = (elements_after / elements_before) if elements_before else 0.0
    char_ratio = (chars_after / chars_before) if chars_before else 0.0

    # Выбираем пороги в зависимости от типа документа
    if is_bilingual:
        # Для билингвального: нормально потерять ~50%, критично потерять >80%
        elem_threshold = 0.20
        char_threshold = 0.25
    else:
        # Для монолингвального: должно оставаться большинство элементов
        elem_threshold = 0.60
        char_threshold = 0.65

    degraded = False
    reason: str | None = None

    if elem_ratio < elem_threshold:
        degraded = True
        reason = (
            f"too_few_elements_retained:{elem_ratio:.2f}"
            f"(threshold={'bilingual' if is_bilingual else 'monolingual'}:{elem_threshold})"
        )
    elif char_ratio < char_threshold:
        degraded = True
        reason = (
            f"too_few_chars_retained:{char_ratio:.2f}"
            f"(threshold={'bilingual' if is_bilingual else 'monolingual'}:{char_threshold})"
        )

    return {
        "elements_before": elements_before,
        "elements_after": elements_after,
        "chars_before": chars_before,
        "chars_after": chars_after,
        "element_retention_ratio": elem_ratio,
        "char_retention_ratio": char_ratio,
        "degraded": degraded,
        "degraded_reason": reason,
    }


async def get_pdf_source_ref(
    document_version_id: str,
    db: AsyncSession,
) -> str | None:
    """
    Получить ссылку на PDF-источник для извлечения.

    Логика:
    - если у версии есть converted_pdf_ref (DOCX/DOC → PDF), используем его;
    - иначе используем original_ref (исходный PDF).
    """
    repo = DocumentVersionRepository(db)
    version = await repo.get_by_id(uuid.UUID(document_version_id))
    if not version:
        return None

    if getattr(version, "converted_pdf_ref", None):
        return version.converted_pdf_ref

    return version.original_ref


async def update_extraction_result(
    document_version_id: str,
    document_id: str,
    raw_ref: str,
    primary_ref: str | None,
    page_count: int,
    strategy: str,
    ocr_confidence: float,
    db: AsyncSession,
) -> None:
    """
    Обновить результаты извлечения в БД (version + document).

    Контракт пайплайна:
    - raw_ref хранит технический multi-language snapshot после extract;
    - layout_primary_ref — единственный downstream layout после extract;
    - все следующие стадии (PII, structure, content/chunks, checks) должны
      использовать именно layout_primary_ref, а не raw_ref.
    """
    ver_repo = DocumentVersionRepository(db)
    doc_repo = DocumentRepository(db)

    version_uuid = uuid.UUID(document_version_id)
    version = await ver_repo.get_by_id(version_uuid)

    pipeline_strategy: str | None = None
    origin_filetype = getattr(version, "origin_filetype", None) if version else None

    if origin_filetype:
        if origin_filetype == "pdf":
            if strategy == "native":
                pipeline_strategy = "pdf_native"
            elif strategy == "ocr":
                pipeline_strategy = "pdf_ocr"
            elif strategy == "hybrid":
                pipeline_strategy = "pdf_hybrid"
        elif origin_filetype == "docx":
            if strategy == "native":
                pipeline_strategy = "docx_via_pdf_native"
            elif strategy == "ocr":
                pipeline_strategy = "docx_via_pdf_ocr"
            elif strategy == "hybrid":
                pipeline_strategy = "docx_via_pdf_hybrid"

    await ver_repo.update_refs(
        version_uuid,
        raw_ref=raw_ref,
        layout_primary_ref=primary_ref,
        extraction_strategy=strategy,
        ocr_confidence=ocr_confidence,
        pipeline_strategy=pipeline_strategy,
    )

    await doc_repo.update_status(
        uuid.UUID(document_id),
        DocumentStatus.EXTRACTED.value,
        page_count=page_count,
        processing_error_details=None,
    )


async def _mark_failed(
    uow: UnitOfWork,
    document_uuid: uuid.UUID,
    *,
    message: str,
    run_id: uuid.UUID | None,
) -> None:
    repo = DocumentRepository(uow.session)
    await repo.set_failed(
        document_uuid,
        stage="extract",
        message=message,
        reason="extraction_error",
        run_id=run_id,
    )


async def _mark_quarantine(
    uow: UnitOfWork,
    document_uuid: uuid.UUID,
    *,
    error: QuarantineError,
    run_id: uuid.UUID | None,
) -> None:
    repo = DocumentRepository(uow.session)
    reason = getattr(error, "reason", None)
    msg = str(error)
    await repo.set_quarantined(
        document_uuid,
        stage="extract",
        message=msg,
        reason=reason,
        run_id=run_id,
    )

"""
PII Sanitizer Worker.

Пайплайн: extract → pii → structure → classify → run_deal_checks.

Назначение:
- Считать одноязычный макет документа (layout_primary) из MinIO.
- Найти PII‑сущности по политике (triggers + YAML‑конфиг типа документа).
- Заменить чувствительные данные токенами, зашифровать mapping и сохранить его.
- Сохранить sanitized‑layout и PII‑mapping в MinIO.
- Обновить статус документа: extracted -> sanitized.
- Запустить следующий этап пайплайна: workers.build_structure.

Архитектурные принципы:
- Celery‑таска синхронная (def) + async‑ядро (PIISanitizer.sanitize) под db_uow().
- Основная санитизация работает в одном UnitOfWork (db_uow) — читаем и пишем всё
  в рамках одной транзакции.
- Если при санитизации происходит ошибка, мы ОТДЕЛЬНО открываем новый db_uow
  только чтобы надёжно записать статус документа (FAILED / QUARANTINE) +
  processing_error_details.
- Для юриста это означает: документ никогда не «зависает» в промежуточном состоянии,
  всегда имеет честный и понятный статус.
"""

import uuid
from typing import Any

from core.config import settings
from core.exceptions import PermanentError, QuarantineError
from core.logging import get_logger
from db.repositories import (
    AuditRepository,
    DocumentRepository,
    DocumentVersionRepository,
)
from db.uow import UnitOfWork, db_uow
from mas_queue.base_task import MASBaseTask
from mas_queue.celery_app import celery_app
from mas_queue.async_helpers import run_async
from services.pii_detector import get_pii_detector, get_preserve_types
from services.pii_encryption import PIIEncryptionError, get_pii_encryptor
from services.pii_tokenizer import PIIPolicy, PIITokenizer
from classifiers.yaml_registry import YAMLRegistry
from core.constants import DocumentStatus
from storage.artifact_store import ArtifactStore, ArtifactPath

logger = get_logger(__name__)


class PIISanitizer:
    """
    Сервис санитизации PII в документе.

    Отделён от Celery‑таски для тестируемости.
    Получает UnitOfWork извне, работает строго в его границах.
    """

    def __init__(
        self,
        uow: UnitOfWork,
        artifact_store,  # ArtifactStore
        detector=None,
        tokenizer=None,
        encryptor=None,
    ):
        self.uow = uow
        self.store = artifact_store

        # Репозитории поверх сессии из UoW
        self.doc_repo = DocumentRepository(uow.session)
        self.version_repo = DocumentVersionRepository(uow.session)
        self.audit_repo = AuditRepository(uow.session)

        # Dependency injection (для тестов)
        self.detector = detector or get_pii_detector()
        self.tokenizer = tokenizer or PIITokenizer()
        self.encryptor = encryptor or get_pii_encryptor()

    async def sanitize(self, document_version_id: uuid.UUID) -> dict[str, Any]:
        """
        Выполнить санитизацию документа.

        Возвращает словарь с полями:
        - document_id
        - entities_detected / tokenized / preserved
        - skipped (True, если документ уже был обработан)

        В этой функции мы НЕ открываем новый UnitOfWork,
        используем только тот, что был передан снаружи.
        """
        # 1. Загрузка версии и документа
        version = await self.version_repo.get_by_id(document_version_id)
        if not version:
            raise PermanentError(f"DocumentVersion {document_version_id} not found")

        doc = await self.doc_repo.get_by_id(version.document_id)
        if not doc:
            raise PermanentError(f"Document {version.document_id} not found")

        # 2. Идемпотентность
        if self._should_skip(doc):
            logger.info(
                "Document already sanitized, skipping",
                document_id=str(doc.id),
            )
            return {"document_id": str(doc.id), "skipped": True}

        # 3. Единый контракт пайплайна:
        # PII работает только по layout_primary_ref.
        layout_ref = getattr(version, "layout_primary_ref", None)
        if not layout_ref:
            raise PermanentError(
                f"layout_primary_ref is required for PII sanitization, version={document_version_id}"
            )

        try:
            layout_raw = self.store.get_json(layout_ref)
            elements = layout_raw.get("elements", [])

            # 5. Построить PIIPolicy с учётом applicable_triggers и YAML‑конфига
            triggers = getattr(doc, "applicable_triggers", None) or []
            policy = self._build_policy(doc, triggers)

            # 6. Обработать элементы
            sanitized_elements, global_mapping, stats = self._process_elements(
                elements, policy
            )

            # 7. Сохранить sanitized-layout.
            #
            # Контракт пайплайна:
            # - layout_primary_sanitized_ref является единственным downstream
            #   sanitized layout;
            # - legacy sanitized layout paths больше не используются.
            layout_sanitized = {**layout_raw, "elements": sanitized_elements}
            sanitized_path = self._build_sanitized_path(doc)
            # ArtifactStore.put_json — синхронный метод, НЕ используем await
            self.store.put_json(sanitized_path, layout_sanitized)

            # 8. Зашифровать и сохранить pii_mapping
            mapping_path = self._build_mapping_path(doc)
            encrypted = self.encryptor.encrypt_mapping(global_mapping)
            # ArtifactStore.put_file — синхронный метод, НЕ используем await
            self.store.put_file(mapping_path, encrypted)

            # 9. Обновить DocumentVersion и статус документа
            #
            # ВАЖНО:
            # - sanitized_path указывает на одноязычный sanitized layout
            #   (ArtifactPath.layout_primary_sanitized).
            # - Через update_refs мы записываем эту ссылку в колонку
            #   layout_primary_sanitized_ref, которая должна быть основным
            #   источником истины для всех следующих стадий (structure,
            #   DocumentContentService, checks).
            await self.version_repo.update_refs(
                version_id=version.id,
                layout_primary_sanitized_ref=sanitized_path,
            )
            await self.doc_repo.update_status(
                doc_id=doc.id,
                status=DocumentStatus.SANITIZED.value,
                processing_error_details=None,
            )

            # 10. AuditLog (опционально)
            if settings.PII_AUDIT_ENABLED:
                await self._write_audit_log(doc, triggers, policy, stats)

            logger.info(
                "PII sanitization complete",
                document_id=str(doc.id),
                entities_detected=stats["detected"],
                entities_tokenized=stats["tokenized"],
                entities_preserved=stats["preserved"],
                role_map_used=policy.use_role_map,
            )

            return {
                "document_id": str(doc.id),
                "entities_detected": stats["detected"],
                "entities_tokenized": stats["tokenized"],
                "entities_preserved": stats["preserved"],
                "skipped": False,
            }

        except PIIEncryptionError as e:
            # Ошибка шифрования → карантин
            raise QuarantineError(
                f"Encryption error: {e}",
                document_id=str(doc.id),
                reason="pii_encryption_error",
            ) from e

    def _should_skip(self, doc) -> bool:
        """Проверить, нужно ли пропустить санитизацию (идемпотентность)."""
        if not settings.PII_SKIP_IF_ALREADY_SANITIZED:
            return False
        return doc.status == DocumentStatus.SANITIZED.value

    def _build_policy(self, doc, triggers: list) -> PIIPolicy:
        """
        Построить PIIPolicy с учётом YAML‑конфига типа документа.
        """
        party_role_keywords: dict[str, str] | None = None

        # На Stage 5 doc_type может быть ещё None (классификация позже)
        type_code = getattr(doc, "doc_type", None)

        if type_code:
            try:
                registry = YAMLRegistry.get_instance()
                cfg = registry.get_config(type_code)

                if cfg and cfg.party_role_keywords:
                    party_role_keywords = cfg.party_role_keywords
                    logger.debug(
                        "Using party_role_keywords from YAML",
                        doc_type=type_code,
                        keywords_count=len(party_role_keywords),
                    )
            except Exception as e:
                logger.warning(
                    "Failed to load YAML config for doc_type",
                    doc_type=type_code,
                    error=str(e),
                )

        return PIITokenizer.build_policy_from_triggers(
            applicable_triggers=triggers,
            party_role_keywords=party_role_keywords,
        )

    def _process_elements(
        self,
        elements: list[dict],
        policy: PIIPolicy,
    ) -> tuple[list[dict], dict[str, str], dict[str, int]]:
        """
        Обработать элементы layout'а.

        Возвращает:
          sanitized_elements, global_mapping, stats
        """
        sanitized_elements = []
        global_mapping: dict[str, str] = {}
        stats = {"detected": 0, "tokenized": 0, "preserved": 0, "role_mapped": 0}

        preserve_types = get_preserve_types()

        for elem in elements:
            raw_text = elem.get("text", "")
            if not raw_text:
                sanitized_elements.append(elem)
                continue

            # Детектируем PII
            all_entities = self.detector.detect(raw_text)
            stats["detected"] += len(all_entities)

            # Оставляем без токенизации типы, которые надо сохранить как есть
            pii_entities = [e for e in all_entities if e.type not in preserve_types]

            if not pii_entities:
                sanitized_elements.append({**elem})
                stats["preserved"] += len(all_entities)
                continue

            # Токенизируем
            context = raw_text[:500]
            result = self.tokenizer.tokenize(
                raw_text,
                pii_entities,
                policy,
                context_window=context,
            )

            global_mapping.update(result.mapping)
            stats["tokenized"] += result.entities_tokenized
            stats["preserved"] += result.entities_preserved
            stats["role_mapped"] += result.entities_role_mapped

            sanitized_elements.append({**elem, "text": result.sanitized_text})

        return sanitized_elements, global_mapping, stats

    def _build_sanitized_path(self, doc) -> str:
        """
        Построить путь для sanitized‑layout в MinIO через ArtifactPath.

        В новой архитектуре это будет одноязычный layout_primary_sanitized.
        """
        # tenant_id берём из связанной сделки, как единственный источник истины
        if hasattr(doc, "deal") and doc.deal and hasattr(doc.deal, "tenant_id"):
            tenant_id = str(doc.deal.tenant_id)
        else:
            # fallback для dev-режима, если связь не подгружена
            tenant_id = str(doc.deal_id)

        deal_id = str(doc.deal_id)
        return ArtifactPath.layout_primary_sanitized(
            tenant_id=tenant_id,
            deal_id=deal_id,
            document_id=str(doc.id),
        )

    def _build_mapping_path(self, doc) -> str:
        """Построить путь для PII‑mapping в MinIO через ArtifactPath."""
        if hasattr(doc, "deal") and doc.deal and hasattr(doc.deal, "tenant_id"):
            tenant_id = str(doc.deal.tenant_id)
        else:
            tenant_id = str(doc.deal_id)

        deal_id = str(doc.deal_id)
        return ArtifactPath.pii_mapping(
            tenant_id=tenant_id,
            deal_id=deal_id,
            document_id=str(doc.id),
        )

    async def _write_audit_log(
        self,
        doc,
        triggers: list,
        policy: PIIPolicy,
        stats: dict,
    ) -> None:
        """Записать audit‑лог по результатам санитизации."""
        if hasattr(doc, "deal") and doc.deal and hasattr(doc.deal, "tenant_id"):
            tenant_id: uuid.UUID = doc.deal.tenant_id
        else:
            # fallback: используем deal_id как tenant_id (dev‑режим)
            tenant_id = doc.deal_id

        await self.audit_repo.log(
            tenant_id=tenant_id,
            deal_id=doc.deal_id,
            action="SANITIZE",
            actor="system:pii_sanitizer",
            actor_role="system",
            details={
                "document_id": str(doc.id),
                "entities_detected": stats["detected"],
                "entities_tokenized": stats["tokenized"],
                "entities_preserved": stats["preserved"],
                "entities_role_mapped": stats["role_mapped"],
                "classification_level": getattr(doc, "classification_level", None),
                "applicable_triggers": triggers,
                "role_map_used": policy.use_role_map,
                "yaml_keywords_used": policy.party_role_keywords is not None,
                "encryption_enabled": self.encryptor.is_encryption_enabled,
            },
        )


@celery_app.task(
    base=MASBaseTask,
    bind=True,
    name="pii.sanitize_document",
    track_job=False,
    queue="default",
)
def sanitize_document(
    self,
    document_version_id: str,
) -> dict:
    """
    Синхронная Celery‑таска санитизации документа.

    Внутри:
      - открывает основной UnitOfWork (uow_main) через db_uow();
      - запускает асинхронную санитизацию через PIISanitizer.sanitize;
      - при ошибке открывает второй, маленький UnitOfWork только для записи
        статуса (FAILED / QUARANTINE) + processing_error_details;
      - при успехе ставит задачу следующего этапа (Structure Builder).
    """
    version_uuid = uuid.UUID(document_version_id)

    async def _run() -> dict:
        # Основной UnitOfWork для санитизации
        async with db_uow() as uow_main:
            store = ArtifactStore(settings)
            sanitizer = PIISanitizer(uow=uow_main, artifact_store=store)

            try:
                result = await sanitizer.sanitize(version_uuid)

                # Успех → ставим следующую стадию (build_structure)
                if (
                    not result.get("skipped")
                    and not settings.DISABLE_DOWNSTREAM_TASKS
                ):
                    from mas_queue.celery_app import celery_app as _celery  # на случай циклов импорта

                    logger.info(
                        "Scheduling build_structure after PII",
                        document_version_id=document_version_id,
                    )

                    _celery.send_task(
                        "workers.build_structure",
                        kwargs={
                            "document_version_id": document_version_id,
                            # run_id можно добавить при необходимости
                        },
                    )

                return result

            except PermanentError as e:
                logger.error(
                    "Permanent error during PII sanitization",
                    document_version_id=document_version_id,
                    error=str(e),
                )
                # Отдельный UoW только для записи статуса FAILED + деталей
                async with db_uow() as uow_status:
                    await _mark_failed(
                        uow_status,
                        version_id=version_uuid,
                        message=str(e),
                    )
                return {
                    "status": DocumentStatus.FAILED.value,
                    "error": str(e),
                }

            except QuarantineError as e:
                logger.error(
                    "Quarantine error during PII sanitization",
                    document_version_id=document_version_id,
                    error=str(e),
                )
                # Отдельный UoW только для записи статуса QUARANTINE + деталей
                async with db_uow() as uow_status:
                    await _mark_quarantine(
                        uow_status,
                        version_id=version_uuid,
                        error=e,
                    )
                return {
                    "status": DocumentStatus.QUARANTINE.value,
                    "error": str(e),
                }

    # Переход в async‑мир через общий мост run_async
    return run_async(_run())


async def _mark_failed(
    uow: UnitOfWork,
    version_id: uuid.UUID,
    *,
    message: str,
) -> None:
    """
    Обновить статус документа на FAILED в отдельной, простой транзакции
    с заполнением processing_error_details.
    """
    version_repo = DocumentVersionRepository(uow.session)
    doc_repo = DocumentRepository(uow.session)

    version = await version_repo.get_by_id(version_id)
    if not version:
        return

    await doc_repo.set_failed(
        version.document_id,
        stage="pii",
        message=message,
        reason="pii_error",
        run_id=None,
    )


async def _mark_quarantine(
    uow: UnitOfWork,
    version_id: uuid.UUID,
    *,
    error: QuarantineError,
) -> None:
    """
    Обновить статус документа на QUARANTINE в отдельной транзакции
    с заполнением processing_error_details.
    """
    version_repo = DocumentVersionRepository(uow.session)
    doc_repo = DocumentRepository(uow.session)

    version = await version_repo.get_by_id(version_id)
    if not version:
        return

    reason = getattr(error, "reason", None)
    await doc_repo.set_quarantined(
        version.document_id,
        stage="pii",
        message=str(error),
        reason=reason,
        run_id=None,
    )

# src/workers/structure_builder.py

"""
Structure Builder Worker.

Назначение:
- Считать санитизированный макет документа (одноязычный layout после PII) из MinIO.
- Построить древовидную структуру документа (clause_tree) через сервис StructureBuilder.
- Сохранить clause_tree в MinIO и обновить ссылки в DocumentVersion.
- Обновить статус документа: sanitized -> structured.
- Запустить следующую задачу пайплайна: workers.classify_document.

Архитектурный паттерн:
- Внешняя функция Celery‑таски остаётся синхронной (def), чтобы Celery
  мог корректно сериализовать результат и работать в prefork‑модели.
- Вся асинхронная логика (БД, MinIO) находится в _build_structure_async()
  и вызывается из таски через безопасный переход sync → async (run_async + db_uow).
"""

import time
import uuid

from core.config import settings
from core.constants import DocumentStatus
from core.exceptions import PermanentError
from core.logging import get_logger
from db.repositories.document_repo import DocumentRepository
from db.repositories.document_version_repo import DocumentVersionRepository
from db.uow import db_uow, UnitOfWork
from mas_queue.base_task import MASBaseTask
from mas_queue.celery_app import celery_app
from mas_queue.async_helpers import run_async
from services.structure.builder import StructureBuilder
from storage.artifact_store import ArtifactPath, ArtifactStore

logger = get_logger(__name__)


@celery_app.task(
    base=MASBaseTask,
    bind=True,
    name="workers.build_structure",
    queue="default",
)
def build_structure(
    self,
    document_version_id: str,
    **kwargs,  # принимаем остальные аргументы (deal_id, run_id и т.д.) для совместимости
) -> dict:
    """
    Sync Celery entry point.

    Паттерн:
    - Celery вызывает синхронную функцию.
    - Внутри один раз передаём управление в async‑мир через run_async.
    """

    async def _wrapper() -> dict:
        try:
            async with db_uow() as uow:
                return await _build_structure_async(
                    document_version_id=document_version_id,
                    extra_args=kwargs,
                    uow=uow,
                )
        except PermanentError as e:
            logger.error(
                "Permanent error during structure building",
                document_version_id=document_version_id,
                error=str(e),
            )
            # Отдельный UoW только для перевода документа в FAILED + деталей
            async with db_uow() as uow_status:
                await _mark_failed(
                    uow_status,
                    document_version_id=document_version_id,
                    message=str(e),
                )
            return {
                "status": DocumentStatus.FAILED.value,
                "document_version_id": document_version_id,
                "error": str(e),
            }

    return run_async(_wrapper())


async def _build_structure_async(
    document_version_id: str,
    extra_args: dict,
    uow: UnitOfWork,
) -> dict:
    store = ArtifactStore(settings)
    start = time.time()
    v_id = (
        uuid.UUID(document_version_id)
        if isinstance(document_version_id, str)
        else document_version_id
    )

    ver_repo = DocumentVersionRepository(uow.session)
    doc_repo = DocumentRepository(uow.session)

    # 1. Загрузка версии и документа
    version = await ver_repo.get_by_id(v_id)
    if not version:
        raise PermanentError(f"DocumentVersion {v_id} not found")

    doc = await doc_repo.get_by_id(version.document_id)
    if not doc:
        raise PermanentError(f"Document {version.document_id} not found")

    # Идемпотентность: не перестраиваем, если уже готово
    if doc.status in (
        DocumentStatus.STRUCTURED.value,
        DocumentStatus.CLASSIFIED.value,
        DocumentStatus.ANALYZING.value,
    ):
        logger.info("Already structured, skipping", document_id=str(doc.id))
        return {"document_id": str(doc.id), "skipped": True}

    # 2. Получение макета из MinIO.
    #
    # Единый контракт пайплайна:
    #   1) layout_primary_sanitized_ref — основной source of truth после PII;
    #   2) layout_primary_ref — допустимый recovery fallback, если PII ещё не выполнен,
    #      но нужно восстановить ход пайплайна.
    layout_ref = (
        getattr(version, "layout_primary_sanitized_ref", None)
        or getattr(version, "layout_primary_ref", None)
    )

    if not layout_ref:
        raise PermanentError(
            f"no layout_primary_sanitized_ref/layout_primary_ref for version {v_id}"
        )

    # ArtifactStore.get_json — синхронный метод, НЕ используем await
    raw_layout = store.get_json(layout_ref)
    elements = raw_layout.get("elements", [])
    if not elements:
        raise PermanentError(f"No elements in layout for version {v_id}")

    # 3. Вызок сервиса построения структуры
    builder = StructureBuilder()
    clause_tree = builder.build(
        document_id=str(doc.id),
        elements=elements,
    )

    # 4. Сохранение ClauseTree как отдельного артефакта
    # ВАЖНО: tenant_id берём из связанной сделки, а не подменяем deal_id.
    if hasattr(doc, "deal") and doc.deal and hasattr(doc.deal, "tenant_id"):
        tenant_id = str(doc.deal.tenant_id)
    else:
        # fallback для dev‑режима, если связь не подгружена
        tenant_id = str(doc.deal_id)

    deal_id = str(doc.deal_id)

    clause_tree_path = ArtifactPath.clause_tree(
        tenant_id=tenant_id,
        deal_id=deal_id,
        document_id=str(doc.id),
    )
    # ArtifactStore.put_json — синхронный метод, НЕ используем await
    store.put_json(clause_tree_path, clause_tree.to_dict())

    # 5. Обновление БД
    await ver_repo.update_refs(version_id=v_id, clause_tree_ref=clause_tree_path)
    await doc_repo.update_status(
        doc.id,
        DocumentStatus.STRUCTURED.value,
        processing_error_details=None,
    )
    # UoW зафиксирует изменения (commit) при выходе из db_uow в таске

    # 6. Запуск следующего этапа (Классификация)
    # Передаём в classify_document достаточный контекст:
    # - run_id (если есть),
    # - deal_id и tenant_id для корректного запуска run_deal_checks через классификатор.
    from mas_queue.celery_app import celery_app as _celery  # локальный импорт на случай циклов

    _celery.send_task(
        "workers.classify_document",
        kwargs={
            "document_id": str(doc.id),
            "run_id": extra_args.get("run_id"),
            "deal_id": deal_id,
            "tenant_id": tenant_id,
        },
    )

    return {
        "document_id": str(doc.id),
        "clause_tree_ref": clause_tree_path,
        "confidence": clause_tree.structure_confidence,
        "is_degraded": clause_tree.degraded,
        "duration": round(time.time() - start, 2),
    }

"""
Run Orchestrator Worker.

Назначение:
- Оркестрация пайплайна анализа для конкретного Run:
  extract -> pii -> structure -> classify -> (далее отдельный воркер run_deal_checks).
- Находит все документы сделки и по их статусу решает, какие задачи поставить:
  - PENDING     -> workers.extract_document
  - EXTRACTED   -> pii.sanitize_document
  - SANITIZED   -> workers.build_structure
  - STRUCTURED  -> workers.classify_document
- Переводит Run в статус RUNNING и запоминает,
  для скольких документов будут запущены последующие стадии.
"""

import uuid

from sqlalchemy.ext.asyncio import AsyncSession

from core.logging import get_logger
from core.constants import DocumentStatus, RunStatus
from db.models.run import Run
from db.repositories.document_repo import DocumentRepository
from db.repositories.document_version_repo import DocumentVersionRepository
from db.repositories.run_repo import RunRepository
from db.uow import db_uow
from mas_queue.base_task import MASBaseTask
from mas_queue.celery_app import celery_app
from mas_queue.async_helpers import run_async  # единый мост sync → async
from services.run_checks_auto_starter import maybe_start_run_checks

logger = get_logger(__name__)


@celery_app.task(
    base=MASBaseTask,
    bind=True,
    name="workers.start_run_pipeline",
    track_job=False,
    queue="default",
)
def start_run_pipeline(
    self,
    run_id: str,
    deal_id: str | None = None,
    tenant_id: str | None = None,
    force_reclassify: bool = False,
) -> dict:
    """
    Sync Celery entrypoint для оркестрации пайплайна Run.

    Шаги:
    1) по run_id находим Run и его deal_id (если не передан);
    2) по deal_id и tenant_id (из сделки) обходим все документы;
    3) для каждого документа по статусу решаем, какие стадии запускать;
    4) переводим Run в статус "running".
    """

    async def _wrapper() -> dict:
        # db_uow создаёт отдельную AsyncSession и управляет commit/rollback/close.
        async with db_uow() as uow:
            return await _start_run_pipeline_async(
                run_id=run_id,
                deal_id=deal_id,
                tenant_id=tenant_id,
                force_reclassify=force_reclassify,
                db=uow.session,
            )

    # Запускаем асинхронный код через единый мост run_async
    return run_async(_wrapper())


async def _start_run_pipeline_async(
    run_id: str,
    deal_id: str | None,
    tenant_id: str | None,
    force_reclassify: bool,
    db: AsyncSession,
) -> dict:
    """
    Async‑ядро оркестрации Run.

    Для непрофессионального программиста:
    - Здесь мы решаем, какие задачи по документам надо запустить.
    - Если документ уже дошёл до нужного статуса, мы можем переиспользовать результат.
    """

    logger.info(
        "Starting run pipeline",
        extra={
            "run_id": run_id,
            "deal_id": deal_id,
            "tenant_id": tenant_id,
            "force_reclassify": force_reclassify,
        },
    )

    run_repo = RunRepository(db)
    doc_repo = DocumentRepository(db)
    ver_repo = DocumentVersionRepository(db)

    run_uuid = uuid.UUID(run_id)
    run_obj: Run | None = await run_repo.get_by_id(run_uuid)
    if not run_obj:
        await run_repo.update_status(
            run_uuid,
            RunStatus.FAILED.value,
            error_details={
                "stage": "orchestrator",
                "type": "config_error",
                "reason": "run_not_found",
                "message": f"Run {run_id} not found",
            },
        )
        logger.error(
            "Run not found for orchestrator",
            extra={"run_id": run_id},
        )
        return {
            "run_id": run_id,
            "deal_id": None,
            "tenant_id": tenant_id,
            "documents_total": 0,
            "tasks_scheduled": 0,
        }

    # Пытаемся атомарно стартовать Run.
    # Если Run уже кто‑то запустил (или он уже завершён),
    # мы не должны повторно ставить задачи по документам.
    started = await run_repo.try_start_run(run_uuid)
    if not started:
        logger.info(
            "Run pipeline already started or finished, skipping orchestration",
            extra={"run_id": run_id},
        )
        return {
            "run_id": run_id,
            "deal_id": str(run_obj.deal_id) if run_obj and run_obj.deal_id else None,
            "tenant_id": tenant_id,
            "documents_total": 0,
            "tasks_scheduled": 0,
        }

    # Определяем стратегию пайплайна для этого Run.
    pipeline_strategy = getattr(run_obj, "pipeline_strategy", "REUSE_STRUCTURE")
    is_full_rerun = pipeline_strategy == "FULL_RERUN"

    logger.info(
        "Resolved run pipeline strategy",
        extra={
            "run_id": run_id,
            "pipeline_strategy": pipeline_strategy,
            "force_reclassify": force_reclassify,
        },
    )

    # deal_id / tenant_id берём из Run / связанной сделки, если не переданы явно.
    deal_uuid = (
        run_obj.deal_id
        if run_obj.deal_id
        else uuid.UUID(deal_id) if deal_id else None
    )
    if deal_uuid is None:
        await run_repo.update_status(
            run_uuid,
            RunStatus.FAILED.value,
            error_details={
                "stage": "orchestrator",
                "type": "config_error",
                "reason": "deal_not_bound",
                "message": f"Run {run_id} has no deal_id",
            },
        )
        logger.error(
            "Run has no deal_id",
            extra={"run_id": run_id},
        )
        return {
            "run_id": run_id,
            "deal_id": None,
            "tenant_id": tenant_id,
            "documents_total": 0,
            "tasks_scheduled": 0,
        }

    deal_id_str = str(deal_uuid)

    if tenant_id is None:
        if hasattr(run_obj, "deal") and run_obj.deal and hasattr(run_obj.deal, "tenant_id"):
            tenant_id_str = str(run_obj.deal.tenant_id)
        else:
            tenant_id_str = deal_id_str
    else:
        tenant_id_str = tenant_id

    # Загружаем все документы сделки.
    docs = await doc_repo.get_by_deal(deal_uuid)
    if not docs:
        logger.info(
            "No documents for deal, nothing to process",
            extra={"run_id": run_id, "deal_id": deal_id_str},
        )
        await run_repo.update_status(run_uuid, RunStatus.COMPLETED.value)

        return {
            "run_id": run_id,
            "deal_id": deal_id_str,
            "tenant_id": tenant_id_str,
            "documents_total": 0,
            "tasks_scheduled": 0,
        }

    tasks_scheduled = 0

    for doc in docs:
        doc_id_str = str(doc.id)
        status = doc.status or ""

        # Берём последнюю версию, чтобы проверить наличие артефактов
        # единого pipeline-контракта (layout_primary / layout_primary_sanitized).
        version = await ver_repo.get_latest(doc.id)

        logger.info(
            "Evaluating document for run pipeline",
            extra={
                "run_id": run_id,
                "document_id": doc_id_str,
                "status": status,
                "doc_type": getattr(doc, "doctype", None),
                "classification_level": getattr(doc, "classification_level", None),
            },
        )

        # PENDING
        if status == DocumentStatus.PENDING.value:
            if not version:
                logger.warning(
                    "Document has PENDING status but no version, skipping extract",
                    extra={"document_id": doc_id_str},
                )
                continue

            celery_app.send_task(
                "workers.extract_document",
                kwargs={
                    "document_version_id": str(version.id),
                    "document_id": doc_id_str,
                    "deal_id": deal_id_str,
                    "tenant_id": tenant_id_str,
                    "run_id": run_id,
                },
            )
            tasks_scheduled += 1
            continue

        # EXTRACTED
        if status == DocumentStatus.EXTRACTED.value:
            if not version or not getattr(version, "layout_primary_ref", None):
                logger.warning(
                    "EXTRACTED document without layout_primary_ref, skipping pii",
                    extra={"document_id": doc_id_str},
                )
                continue

            celery_app.send_task(
                "pii.sanitize_document",
                kwargs={
                    "document_version_id": str(version.id),
                },
            )
            tasks_scheduled += 1
            continue

        # SANITIZED
        if status == DocumentStatus.SANITIZED.value:
            if not version:
                logger.warning(
                    "SANITIZED document without version, skipping structure",
                    extra={"document_id": doc_id_str},
                )
                continue

            has_sanitized = bool(
                getattr(version, "layout_primary_sanitized_ref", None)
            )
            has_primary = bool(getattr(version, "layout_primary_ref", None))

            # Для непрофессионального программиста:
            # - Нормальный путь: есть layout_primary_sanitized_ref (после PII).
            # - Recovery-путь: если по какой-то причине PII не сохранил sanitized layout,
            #   но есть layout_primary_ref, StructureBuilder по контракту умеет
            #   использовать и его.
            if not has_sanitized and not has_primary:
                logger.warning(
                    "SANITIZED document without layout_primary_sanitized_ref "
                    "or layout_primary_ref, skipping structure",
                    extra={"document_id": doc_id_str},
                )
                continue

            if not has_sanitized and has_primary:
                logger.warning(
                    "SANITIZED document without sanitized layout, "
                    "falling back to layout_primary_ref for structure building",
                    extra={"document_id": doc_id_str},
                )

            celery_app.send_task(
                "workers.build_structure",
                kwargs={
                    "document_version_id": str(version.id),
                    "run_id": run_id,
                },
            )
            tasks_scheduled += 1
            continue

        # STRUCTURED
        if status == DocumentStatus.STRUCTURED.value:
            celery_app.send_task(
                "workers.classify_document",
                kwargs={
                    "document_id": doc_id_str,
                    "run_id": run_id,
                    "deal_id": deal_id_str,
                    "tenant_id": tenant_id_str,
                },
            )
            tasks_scheduled += 1
            continue

        # CLASSIFIED: уже классифицирован
        if status == DocumentStatus.CLASSIFIED.value:
            if is_full_rerun or force_reclassify:
                # Вариант "полного прогона" — заново запускаем классификацию.
                celery_app.send_task(
                    "workers.classify_document",
                    kwargs={
                        "document_id": doc_id_str,
                        "run_id": run_id,
                        "deal_id": deal_id_str,
                        "tenant_id": tenant_id_str,
                    },
                )
                tasks_scheduled += 1
            else:
                logger.info(
                    "Document already classified, reusing classification for this run",
                    extra={"document_id": doc_id_str, "run_id": run_id},
                )
            continue

        # COMPLETED: документ уже прошёл полный пайплайн в прошлом Run.
        if status == DocumentStatus.COMPLETED.value:
            if is_full_rerun:
                celery_app.send_task(
                    "workers.classify_document",
                    kwargs={
                        "document_id": doc_id_str,
                        "run_id": run_id,
                        "deal_id": deal_id_str,
                        "tenant_id": tenant_id_str,
                    },
                )
                tasks_scheduled += 1
            else:
                logger.info(
                    "Document already completed, reusing for this run",
                    extra={"document_id": doc_id_str, "run_id": run_id},
                )
            continue

        # FAILED / QUARANTINE / другие статусы
        # ВАЖНО:
        # - Документы в статусах FAILED / QUARANTINE мы здесь не трогаем.
        # - Они не участвуют в дальнейших стадиях и не блокируют запуск checks,
        #   потому что maybe_start_run_checks учитывает только "живые" документы.
        # - Если оставить документ навсегда в промежуточном статусе
        #   (например, structured), но не перевести в FAILED/QUARANTINE
        #   при перманентной ошибке, Run может зависнуть в RUNNING.

        logger.info(
            "Document in unsupported status for run pipeline, skipping",
            extra={"document_id": doc_id_str, "status": status, "run_id": run_id},
        )

    logger.info(
        "Run pipeline documents evaluation complete",
        extra={
            "run_id": run_id,
            "deal_id": deal_id_str,
            "tenant_id": tenant_id_str,
            "documents_total": len(docs),
            "tasks_scheduled": tasks_scheduled,
        },
    )

    logger.info(
        "Run marked as RUNNING, calling maybe_start_run_checks",
        extra={
            "run_id": run_id,
            "deal_id": deal_id_str,
        },
    )

    # После того как Run стал RUNNING (в начале функции через try_start_run),
    # даём сигнал авто-стартеру checks.
    # Для непрофессионального программиста:
    # - maybe_start_run_checks сам посмотрит все документы сделки;
    # - если все "живые" документы уже CLASSIFIED/COMPLETED и
    #   ещё не запускались checks, он один раз поставит задачу
    #   workers.run_deal_checks.
    await maybe_start_run_checks(db=db, run_id=run_uuid)

    logger.info(
        "Run pipeline scheduling complete",
        extra={
            "run_id": run_id,
            "deal_id": deal_id_str,
            "tenant_id": tenant_id_str,
            "documents_total": len(docs),
            "tasks_scheduled": tasks_scheduled,
        },
    )

    return {
        "run_id": run_id,
        "deal_id": deal_id_str,
        "tenant_id": tenant_id_str,
        "documents_total": len(docs),
        "tasks_scheduled": tasks_scheduled,
    }

"""
Abstraction over MinIO/S3 for storing and retrieving artifacts.

All pipeline artifacts (PDFs, JSON layouts, results) are stored here.
Components exchange S3 URIs (references), not raw files.
"""

import io
import json
from urllib.parse import quote
from datetime import timedelta
from minio import Minio
from minio.error import S3Error
from .exports import ExportSuffix, build_export_filename

from core.config import Settings, settings
from core.exceptions import PermanentError, TransientError

class ArtifactPath:
    """
    Builds S3 paths according to the specification:
    /tenants/{tenant_id}/deals/{deal_id}/...
    """

    @staticmethod
    def original(tenant_id: str, deal_id: str, document_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/originals/{document_id}"

    @staticmethod
    def converted_pdf(tenant_id: str, deal_id: str, document_id: str) -> str:
        """
        Путь к PDF-версии документа, сконвертированной из офисного формата (DOCX, DOC и т.п.).

        Используется в convert_worker, когда исходный файл не PDF:
        - original_ref хранит ссылку на DOCX-оригинал,
        - converted_pdf_ref хранит ссылку на этот PDF.

        Пример пути:
        tenants/abc/deals/xyz/originals/{document_id}_converted.pdf
        """
        return f"tenants/{tenant_id}/deals/{deal_id}/originals/{document_id}_converted.pdf"

    @staticmethod
    def layout_raw(tenant_id: str, deal_id: str, document_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/layouts/{document_id}/raw.json"

    @staticmethod
    def layout_sanitized(tenant_id: str, deal_id: str, document_id: str) -> str:
        """
        Legacy path for pre-primary sanitized layout.

        Do not use in the active pipeline.
        Kept temporarily only for backward compatibility / migration.
        """
        return f"tenants/{tenant_id}/deals/{deal_id}/layouts/{document_id}/sanitized.json"

    @staticmethod
    def clause_tree(tenant_id: str, deal_id: str, document_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/structures/{document_id}/clause_tree.json"

    @staticmethod
    def check_results(tenant_id: str, deal_id: str, run_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/results/{run_id}/check_results.jsonl"

    @staticmethod
    def evidence(tenant_id: str, deal_id: str, run_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/results/{run_id}/evidence.jsonl"

    @staticmethod
    def risk_summary(tenant_id: str, deal_id: str, run_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/results/{run_id}/risk_summary.json"

    @staticmethod
    def export(tenant_id: str, deal_id: str, export_id: str, filename: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/exports/{export_id}/{filename}"

    @staticmethod
    def pii_mapping(tenant_id: str, deal_id: str, document_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/pii/{document_id}/mapping.enc"

    @staticmethod
    def embeddings(tenant_id: str, deal_id: str, document_id: str) -> str:
        return f"tenants/{tenant_id}/deals/{deal_id}/embeddings/{document_id}.npy"

    @staticmethod
    def document_type_registry() -> str:
        return "playbooks/document_types/registry.yaml"

    @staticmethod
    def document_type_config(doc_type_code: str) -> str:
        return f"playbooks/document_types/{doc_type_code}.yaml"

    @staticmethod
    def playbook_config(playbook_name: str, version: str) -> str:
        return f"playbooks/{playbook_name}/{version}/config.yaml"

    @staticmethod
    def layout_primary(tenant_id: str, deal_id: str, document_id: str) -> str:
        """
        Одноязычный layout под выбранный основной язык сделки (Deal.primary_language).

        Формируется на основе layout_raw фильтрацией по языку.
        Используется всеми downstream-стадиями (PII, структура, LLM и т.п.).
        """
        return f"tenants/{tenant_id}/deals/{deal_id}/layouts/{document_id}/primary.json"

    @staticmethod

class ArtifactStore:
    """
    MinIO/S3 client for artifact storage.
    """

    def __init__(self, app_settings: Settings | None = None):
        s = app_settings or settings
        self.client = Minio(
            endpoint=s.MINIO_ENDPOINT,
            access_key=s.MINIO_ACCESS_KEY,
            secret_key=s.MINIO_SECRET_KEY,
            secure=s.MINIO_SECURE,
        )
        self.bucket = s.MINIO_BUCKET
        self._ensure_bucket()

        self._endpoint = s.MINIO_ENDPOINT
        self._secure = s.MINIO_SECURE
        

    def _ensure_bucket(self) -> None:
        """Create bucket if it does not exist."""
        try:
            if not self.client.bucket_exists(self.bucket):
                self.client.make_bucket(self.bucket)
                logger.info("Created MinIO bucket", bucket=self.bucket)
        except S3Error as e:
            raise TransientError(
                f"Failed to ensure bucket '{self.bucket}': {e}",
                details={"bucket": self.bucket, "error": str(e)},
            )

    def put_json(self, path: str, data: dict | list) -> str:
        """Serialize data to JSON and upload to MinIO."""
        try:
            json_bytes = json.dumps(data, ensure_ascii=False, indent=2).encode("utf-8")
            data_stream = io.BytesIO(json_bytes)

            self.client.put_object(
                bucket_name=self.bucket,
                object_name=path,
                data=data_stream,
                length=len(json_bytes),
                content_type="application/json",
            )

            uri = f"s3://{self.bucket}/{path}"
            logger.debug("Stored JSON artifact", path=path, size_bytes=len(json_bytes))
            return uri
        except S3Error as e:
            raise TransientError(f"Failed to store JSON at '{path}': {e}")

    def get_json(self, path: str) -> dict | list:
        """Download JSON artifact from MinIO and deserialize."""
        try:
            # Очистка пути от префикса s3:// если он передан
            clean_path = path.replace(f"s3://{self.bucket}/", "")

            response = self.client.get_object(
                bucket_name=self.bucket,
                object_name=clean_path,
            )
            json_bytes = response.read()
            response.close()
            response.release_conn()
            return json.loads(json_bytes.decode("utf-8"))
        except S3Error as e:
            if e.code == "NoSuchKey":
                raise PermanentError(f"Artifact not found: '{path}'")
            raise TransientError(f"Failed to get JSON from '{path}': {e}")

    def put_file(
        self,
        path: str,
        file_data: bytes | io.BytesIO,
        content_type: str = "application/octet-stream",
    ) -> str:
        """Upload binary file to MinIO."""
        try:
            if isinstance(file_data, bytes):
                data_stream = io.BytesIO(file_data)
                length = len(file_data)
            else:
                data_stream = file_data
                current_pos = data_stream.tell()
                data_stream.seek(0, 2)
                length = data_stream.tell()
                data_stream.seek(current_pos)

            self.client.put_object(
                bucket_name=self.bucket,
                object_name=path,
                data=data_stream,
                length=length,
                content_type=content_type,
            )
            return f"s3://{self.bucket}/{path}"
        except S3Error as e:
            raise TransientError(f"Failed to store file at '{path}': {e}")

    def get_file(self, path: str) -> bytes:
        """Download binary file from MinIO."""
        try:
            clean_path = path.replace(f"s3://{self.bucket}/", "")
            response = self.client.get_object(
                bucket_name=self.bucket,
                object_name=clean_path,
            )
            data = response.read()
            response.close()
            response.release_conn()
            return data
        except S3Error as e:
            if e.code == "NoSuchKey":
                raise PermanentError(f"Artifact not found: '{path}'")
            raise TransientError(f"Failed to get file: {e}")


    def get_http_url(self, path: str) -> str:
        """
        Вернуть HTTP/HTTPS URL для объекта в MinIO по s3://... или относительному пути.

        На dev-стенде этого достаточно, чтобы браузер смог открыть/скачать файл.
        При желании можно заменить реализацию на presigned URL.
        """
        # убрать префикс s3://bucket/ если он есть
        clean_path = path.replace(f"s3://{self.bucket}/", "")

        # URL-encode object name, чтобы пробелы и спецсимволы не ломали ссылку
        object_name = quote(clean_path)

        scheme = "https" if self._secure else "http"
        return f"{scheme}://{self._endpoint}/{self.bucket}/{object_name}"


    def get_presigned_get_url(self, path: str, expires_seconds: int = 3600) -> str:
        """
        Вернуть presigned URL для скачивания объекта.

        path может быть как s3://bucket/..., так и относительным путём.
        """
        clean_path = path.replace(f"s3://{self.bucket}/", "")

        # Minio ждет именно timedelta, а не int
        expires = timedelta(seconds=expires_seconds)

        return self.client.presigned_get_object(
            bucket_name=self.bucket,
            object_name=clean_path,
            expires=expires,
        )


    def exists(self, path: str) -> bool:
        """Check if an object exists at the given path."""
        try:
            clean_path = path.replace(f"s3://{self.bucket}/", "")
            self.client.stat_object(bucket_name=self.bucket, object_name=clean_path)
            return True
        except S3Error as e:
            if e.code == "NoSuchKey":
                return False
            raise TransientError(f"Failed to check existence: {e}")

    def delete(self, path: str) -> None:
        """Delete an object at the given path."""
        try:
            clean_path = path.replace(f"s3://{self.bucket}/", "")
            self.client.remove_object(bucket_name=self.bucket, object_name=clean_path)
        except S3Error as e:
            raise TransientError(f"Failed to delete artifact: {e}")

    def save_export_artifact(
        self,

        deal_id: str,

        if content_type == "application/json":
            ext = "json"
        elif content_type == "text/csv":
            ext = "csv"
        else:
            ext = "bin"

        filename = build_export_filename(run_id, suffix, ext=ext)
        path = ArtifactPath.export(tenant_id, deal_id, run_id, filename)

"""
Layout language filter.

Фильтрует layout (text_layout_raw) до одного целевого языка.

Два режима работы:
- Простой (monolingual или короткий документ): посимвольная детекция скрипта,
  оставляем элементы целевого языка + нейтральные короткие блоки.
- Структурный (bilingual, ≥10 элементов): StructuralAnalyzer группирует
  параллельные блоки по вертикальной позиции и номерам пунктов,
  затем из каждой пары выбирается нужный язык.

Форматы:

  ВХОД (text_layout_raw):
  {
    "document_id": str,
    "document_version_id": str,
    "elements": [
      {
        "element_id": str,        # обязательно
        "text": str,              # обязательно
        "type": str,              # "text" | "heading" | "table" | ...
        "page": int,
        "bbox": [x0, y0, x1, y1],
        "clause_uid": str | None,     # может отсутствовать на этой стадии
        ...                           # любые другие поля — сохраняются as-is
      }
    ],
    "tables": [...],
    "metadata": {...}
  }

  ВЫХОД (layout_primary):
  {
    "document_id": str,
    "document_version_id": str,
    "elements": [
      {
        # Все поля оригинального элемента сохранены.
        # Добавлены/обновлены:
        "clause_uid": str | None,         # если структурный анализ нашёл номер
        "parent_clause_uid": str | None,  # если блок является телом пункта
      }
    ],
    "tables": [...],
    "metadata": {
      "language_filter": {
        "target_language": str,
        "bilingual_mode": bool,
        "structural_filter_used": bool,
        "elements_before": int,
        "elements_after": int,
        "chars_before": int,
        "chars_after": int,
        "element_retention_ratio": float,
        "char_retention_ratio": float,
        "degraded": bool,
        "degraded_reason": str | None,
      },
      "element_count": int,
      ...  # остальные поля metadata сохранены
    }
  }

  Downstream (structure_builder) ожидает из elements:
  - element_id (str)
  - text (str)
  - page (int)
  - type (str) — для распознавания heading
  - clause_uid — подсказка для построения дерева (необязательно)
  - parent_clause_uid — подсказка для построения дерева (необязательно)

# Константы детекции языка
# ---------------------------------------------------------------------------

# Порог доли символов нужного скрипта для однозначной классификации
_SCRIPT_DOMINANCE = 0.70
# Минимальное число символов скрипта для классификации (защита от "A" → "en")
_SCRIPT_MIN_CHARS = 3
# Нейтральный блок (цифры, пунктуация) включается только если короткий
_NEUTRAL_MAX_CHARS = 40
# Минимальная длина текста для структурного анализа (чтобы не анализировать мусор)
_STRUCT_MIN_TEXT_LEN = 5
# Y-допуск при группировке параллельных блоков (пиксели/pt PDF)
_Y_TOLERANCE = 8.0
# Минимальный размер документа для включения структурного анализа
_STRUCTURAL_MIN_ELEMENTS = 10


# ---------------------------------------------------------------------------
# Детекция скрипта
# ---------------------------------------------------------------------------

def _detect_script(text: str) -> str:
    """
    Определить доминирующий скрипт строки.

    Возвращает:
      "ru"    — доминирует кириллица
      "en"    — доминирует латиница
      "other" — нейтральный (цифры, пунктуация, пустая строка)

    ВАЖНО: метод намеренно НЕ возвращает "mixed".
    Если оба скрипта присутствуют, побеждает тот, которого больше.
    Это позволяет корректно обработать строки вида "Griffin-Autoservice ООО"
    (название компании с латинским и кириллическим текстом).
    """
    if not text:
        return "other"

    cyr = 0
    lat = 0

    for ch in text:
        code = ord(ch)
        # Кириллица: основной блок + расширенный
        if 0x0400 <= code <= 0x04FF or 0x0500 <= code <= 0x052F:
            cyr += 1
        # Латиница: ASCII + латинское дополнение
        elif (
            0x0041 <= code <= 0x005A
            or 0x0061 <= code <= 0x007A
            or 0x00C0 <= code <= 0x00FF
        ):
            lat += 1

    total = cyr + lat
    if total == 0:
        return "other"

    if cyr >= _SCRIPT_MIN_CHARS and cyr / total >= _SCRIPT_DOMINANCE:
        return "ru"
    if lat >= _SCRIPT_MIN_CHARS and lat / total >= _SCRIPT_DOMINANCE:
        return "en"

    # Оба скрипта присутствуют, но ни один не доминирует.
    # Отдаём доминирующему, а не выбрасываем — см. docstring.
    if cyr >= _SCRIPT_MIN_CHARS or lat >= _SCRIPT_MIN_CHARS:
        return "ru" if cyr >= lat else "en"

    return "other"

# Вспомогательные функции
# ---------------------------------------------------------------------------

def _count_chars(text: str) -> int:
    """Число символов без пробелов и переносов."""
    return len(text.replace(" ", "").replace("\n", "")) if text else 0


def _should_keep_neutral(text: str) -> bool:
    """
    Нейтральный блок (числа, знаки, короткие подписи) включаем только если он
    короткий — это номера пунктов, даты, подписи.
    Длинный нейтральный текст — скорее всего ошибка детекции, пропускаем.
    """
    return _count_chars(text) <= _NEUTRAL_MAX_CHARS

def detect_languages_in_layout(layout: Dict[str, Any]) -> List[str]:
    """
    Определить, какие языки реально присутствуют в layout.

    Анализирует все elements layout["elements"].
    Возвращает отсортированный список кодов языков (["en", "ru"] и т.п.).
    Если ни одного языка не обнаружено — возвращает ["ru"] как дефолт.
    """
    elements: List[Dict[str, Any]] = layout.get("elements") or []
    langs: set[str] = set()

    for el in elements:
        text = (el.get("text") or "").strip()
        if not text:
            continue
        script = _detect_script(text)
        if script in ("ru", "en"):
            langs.add(script)

    return sorted(langs) if langs else ["ru"]

@dataclass
class _StructElement:
    """Внутреннее представление одного элемента layout."""
    original: Dict[str, Any]
    element_id: str
    text: str
    page: int
    bbox: List[float]
    element_type: str
    detected_lang: str         # "ru" | "en" | "other"
    clause_number: Optional[str] = None
    y_center: float = 0.0
    x_center: float = 0.0

    def __post_init__(self) -> None:
        if self.bbox and len(self.bbox) >= 4:
            try:
                floats = [float(v) for v in self.bbox[:4]]
                self.y_center = (floats[1] + floats[3]) / 2.0
                self.x_center = (floats[0] + floats[2]) / 2.0
            except (ValueError, TypeError):
                self.y_center = 0.0
                self.x_center = 0.0
        else:
            self.y_center = float("inf")
            self.x_center = 0.0


@dataclass
class _StructBlock:
    """
    Группа параллельных элементов на одной вертикальной позиции.
    Может содержать один или два языка (ru и/или en).
    """
    block_id: Optional[str]     # номер пункта, если блок является заголовком
    y_position: float
    page: int
    elements: List[_StructElement] = field(default_factory=list)
    ru_element: Optional[_StructElement] = None
    en_element: Optional[_StructElement] = None
    is_heading: bool = False

    def resolve_languages(self) -> None:
        """Выбрать по одному представителю для каждого языка."""
        ru_candidates: List[_StructElement] = []
        en_candidates: List[_StructElement] = []

        for el in self.elements:
            if el.detected_lang == "ru":
                ru_candidates.append(el)
            elif el.detected_lang == "en":
                en_candidates.append(el)
            # "other" добавляем к тому языку, который уже есть в блоке,
            # или оставляем для финального решения

        # Выбираем лучшего кандидата: приоритет — heading, затем самый длинный текст
        def _best(candidates: List[_StructElement]) -> Optional[_StructElement]:
            if not candidates:
                return None
            headings = [e for e in candidates if e.element_type == "heading"]
            pool = headings if headings else candidates
            return max(pool, key=lambda e: len(e.text))

        self.ru_element = _best(ru_candidates)
        self.en_element = _best(en_candidates)

        # Если блок полностью нейтральный (только "other"), назначаем его обоим языкам
        if self.ru_element is None and self.en_element is None:
            neutral = [e for e in self.elements if e.detected_lang == "other"]
            if neutral:
                best_neutral = max(neutral, key=lambda e: len(e.text))
                self.ru_element = best_neutral
                self.en_element = best_neutral

    def get_for_language(self, lang: str) -> Optional[_StructElement]:
        """
        Вернуть элемент для целевого языка.
        Fallback: если нужного языка нет, берём другой (нейтральный контент).
        """
        lang = lang.lower()
        if lang == "ru":
            return self.ru_element or self.en_element
        if lang == "en":
            return self.en_element or self.ru_element
        return self.ru_element or self.en_element

    @property
    def has_both_languages(self) -> bool:
        return self.ru_element is not None and self.en_element is not None


# Паттерны для распознавания номеров пунктов в заголовках
_CLAUSE_PATTERNS = [
    re.compile(r"^(\d+(?:\.\d+)*)\s*[.)]\s*(.+)?$"),
    re.compile(r"^(\d+)\s*[.)]\s*(.+)?$"),
    re.compile(r"^(I{1,3}|IV|V?I{0,3})[.)]\s*(.+)?$", re.IGNORECASE),
    re.compile(r"^([A-Fa-f])[.)]\s*(.+)?$"),
]


class _StructuralAnalyzer:
    """
    Билингвальный структурный анализатор.

    Группирует элементы layout по вертикальной позиции (page, y_center),
    выравнивает параллельные блоки по номерам пунктов,
    и для каждого блока предоставляет выбор нужного языка.
    """

    def __init__(self, y_tolerance: float = _Y_TOLERANCE):
        self.y_tolerance = y_tolerance

    def analyze(self, elements: List[Dict[str, Any]]) -> List[_StructBlock]:
        """Основная точка входа."""
        if not elements:
            return []

        struct_els = self._convert_elements(elements)
        if not struct_els:
            return []

        y_groups = self._group_by_y(struct_els)
        sorted_keys = sorted(y_groups.keys())  # (int, float) — безопасная сортировка

        blocks: List[_StructBlock] = []
        for key in sorted_keys:
            block = self._create_block(y_center=key[1], page=key[0], elements=y_groups[key])
            if block:
                blocks.append(block)

        blocks = self._align_by_clause_numbers(blocks)
        return self._finalize_blocks(blocks)

    def _convert_elements(self, elements: List[Dict[str, Any]]) -> List[_StructElement]:
        result: List[_StructElement] = []
        for el in elements:
            text = el.get("text", "")
            if len(text) < _STRUCT_MIN_TEXT_LEN:
                continue

            # Агрессивное приведение типов — page всегда int
            raw_page = el.get("page", 1)
            try:
                page_val = int(float(str(raw_page).strip()))
            except (ValueError, TypeError, AttributeError):
                page_val = 1

            # bbox всегда list[float]
            raw_bbox = el.get("bbox", [0, 0, 0, 0])
            safe_bbox: List[float] = [0.0, 0.0, 0.0, 0.0]
            try:
                if isinstance(raw_bbox, (list, tuple)) and len(raw_bbox) >= 4:
                    safe_bbox = [float(str(v).strip()) for v in raw_bbox[:4]]
            except (ValueError, TypeError, AttributeError):
                pass

            el_type = el.get("type", "text")
            detected_lang = _detect_script(text)
            clause_num = self._extract_clause_number(text, el_type)

            result.append(_StructElement(
                original=el,
                element_id=str(el.get("element_id", "")),
                text=text,
                page=page_val,
                bbox=safe_bbox,
                element_type=el_type,
                detected_lang=detected_lang,
                clause_number=clause_num,
            ))
        return result

    @staticmethod
    def _extract_clause_number(text: str, el_type: str) -> Optional[str]:
        is_heading = el_type == "heading" or text.strip().isupper()
        stripped = text.strip()
        if not is_heading:
            # Для не-заголовков проверяем только если строка начинается как пункт
            if not any(p.match(stripped) for p in _CLAUSE_PATTERNS):
                return None
        for pattern in _CLAUSE_PATTERNS:
            m = pattern.match(stripped)
            if m:
                return m.group(1)
        return None

    def _group_by_y(
        self, elements: List[_StructElement]
    ) -> Dict[Tuple[int, float], List[_StructElement]]:
        """Группировка по (page, y_center) с допуском y_tolerance."""
        groups: Dict[Tuple[int, float], List[_StructElement]] = {}
        current_key: Optional[Tuple[int, float]] = None

        sorted_els = sorted(elements, key=lambda e: (e.page, e.y_center))

        for el in sorted_els:
            if (
                current_key is not None
                and current_key[0] == el.page
                and abs(el.y_center - current_key[1]) <= self.y_tolerance
            ):
                groups[current_key].append(el)
            else:
                current_key = (el.page, el.y_center)
                groups[current_key] = [el]

        return groups

    def _create_block(
        self,
        y_center: float,
        page: int,
        elements: List[_StructElement],
    ) -> Optional[_StructBlock]:
        if not elements:
            return None

        block = _StructBlock(
            block_id=None,
            y_position=y_center,
            page=page,
            elements=elements,
        )
        block.resolve_languages()

        # Блок является заголовком если есть heading-элемент с номером пункта
        headings_with_num = [
            e for e in elements
            if e.element_type == "heading" and e.clause_number
        ]
        if headings_with_num:
            block.block_id = headings_with_num[0].clause_number
            block.is_heading = True

        return block

    def _align_by_clause_numbers(self, blocks: List[_StructBlock]) -> List[_StructBlock]:
        """
        Объединяем блоки с одинаковым block_id (номером пункта).
        Это нужно для случаев, когда параллельные блоки договора
        оказались на слегка разных y-координатах.
        """
        num_to_blocks: Dict[str, List[_StructBlock]] = {}
        unnumbered: List[_StructBlock] = []

        for b in blocks:
            if b.block_id:
                num_to_blocks.setdefault(b.block_id, []).append(b)
            else:
                unnumbered.append(b)

        merged: List[_StructBlock] = []
        for num, blks in num_to_blocks.items():
            if len(blks) == 1:
                merged.append(blks[0])
            else:
                merged.append(self._merge_blocks(num, blks))

        merged.extend(unnumbered)
        return merged

    @staticmethod
    def _merge_blocks(number: str, blocks: List[_StructBlock]) -> _StructBlock:
        avg_y = sum(b.y_position for b in blocks) / len(blocks)
        all_els = [e for b in blocks for e in b.elements]
        result = _StructBlock(
            block_id=number,
            y_position=avg_y,
            page=blocks[0].page,
            elements=all_els,
        )
        result.resolve_languages()
        result.is_heading = any(b.is_heading for b in blocks)
        return result

    @staticmethod
    def _finalize_blocks(blocks: List[_StructBlock]) -> List[_StructBlock]:
        """Сортируем и дедуплицируем элементы внутри блоков."""
        blocks.sort(key=lambda b: (b.page, b.y_position))
        seen_ids: set[str] = set()
        for b in blocks:
            unique: List[_StructElement] = []
            for e in b.elements:
                if e.element_id not in seen_ids:
                    seen_ids.add(e.element_id)
                    unique.append(e)
            b.elements = unique
        return blocks


# ---------------------------------------------------------------------------
# Оценка качества фильтрации
# ---------------------------------------------------------------------------

def _assess_quality(
    original: List[Dict[str, Any]],
    selected: List[Dict[str, Any]],
) -> Dict[str, Any]:
    e_b = len(original)
    e_a = len(selected)
    c_b = sum(_count_chars(el.get("text", "")) for el in original)
    c_a = sum(_count_chars(el.get("text", "")) for el in selected)

    return {
        "elements_before": e_b,
        "elements_after": e_a,
        "chars_before": c_b,
        "chars_after": c_a,
        "element_retention_ratio": e_a / e_b if e_b else 0.0,
        "char_retention_ratio": c_a / c_b if c_b else 0.0,
        "degraded": False,
        "degraded_reason": None,
    }


def _build_metadata(
    layout: Dict[str, Any],
    target: str,
    quality: Dict[str, Any],
    is_bilingual: bool = False,
    structural_filter_used: bool = False,
) -> Dict[str, Any]:
    """
    Собрать metadata для layout_primary.

    Добавляем language_filter секцию с флагами bilingual_mode и
    structural_filter_used — они используются в extract_worker для
    корректного выбора порогов degraded, и в structure_builder для
    понимания что уже сделано.
    """
    base = dict(layout.get("metadata") or {})
    base["language_filter"] = {
        "target_language": target,
        "bilingual_mode": is_bilingual,
        "structural_filter_used": structural_filter_used,
        **quality,
    }
    base["element_count"] = quality["elements_after"]
    return base


# ---------------------------------------------------------------------------
# Простой фильтр (монолингвальный / short document fallback)
# ---------------------------------------------------------------------------

def _simple_filter(
    layout: Dict[str, Any],
    target: str,
    elements: List[Dict[str, Any]],
    tables: List[Any],
    is_bilingual: bool = False,
) -> Dict[str, Any]:
    """
    Простая посимвольная фильтрация по скрипту.

    Правила:
    - Элемент целевого языка → включаем всегда.
    - Нейтральный "other" → включаем только если текст короткий
      (_NEUTRAL_MAX_CHARS). Длинный нейтральный = ошибка детекции, пропускаем.
    - НЕ включаем элементы другого языка (ни при каких условиях).

    ВАЖНО: "mixed" здесь не возникает, т.к. _detect_script больше не
    возвращает "mixed" — он всегда разрешает в один из языков или "other".
    """
    filtered: List[Dict[str, Any]] = []

    for el in elements:
        text = el.get("text", "")
        lang = _detect_script(text)

        if lang == target:
            filtered.append(el)
        elif lang == "other" and _should_keep_neutral(text):
            # Нейтральные короткие блоки (номера, даты, подписи) включаем
            filtered.append(el)
        # lang == другой язык → пропускаем

    quality = _assess_quality(elements, filtered)
    metadata = _build_metadata(
        layout, target, quality,
        is_bilingual=is_bilingual,
        structural_filter_used=False,
    )

    return {
        "document_version_id": layout.get("document_version_id"),
        "document_id": layout.get("document_id"),
        "elements": filtered,
        "tables": tables,
        "metadata": metadata,
    }


# ---------------------------------------------------------------------------
# Структурный фильтр (билингвальный режим)
# ---------------------------------------------------------------------------

def _structural_filter(
    layout: Dict[str, Any],
    target: str,
    elements: List[Dict[str, Any]],
    tables: List[Any],
) -> Dict[str, Any]:
    """
    Структурный фильтр для билингвальных документов.

    Использует _StructuralAnalyzer для группировки параллельных блоков
    и выбора нужного языка из каждой пары.

    Добавляет к каждому элементу:
    - clause_uid — если блок является заголовком пункта
    - parent_clause_uid — если блок является телом пункта

    Эти поля используются structure_builder как подсказки при построении
    clause_tree, что резко улучшает качество структуры.
    """
    analyzer = _StructuralAnalyzer()
    blocks = analyzer.analyze(elements)

    selected_elements: List[Dict[str, Any]] = []
    last_heading_uid: Optional[str] = None
    bilingual_block_count = 0

    for block in blocks:
        if block.has_both_languages:
            bilingual_block_count += 1

        chosen = block.get_for_language(target)
        if not chosen:
            continue

        # Копируем оригинальный элемент (сохраняем все поля)
        out = dict(chosen.original)
        out["text"] = chosen.text

        # Добавляем структурные подсказки
        if block.is_heading and block.block_id:
            out["clause_uid"] = block.block_id
            last_heading_uid = block.block_id
        elif not block.is_heading and last_heading_uid:
            out["parent_clause_uid"] = last_heading_uid

        selected_elements.append(out)

    # Сортируем по странице и y-позиции
    selected_elements.sort(
        key=lambda e: (
            e.get("page", 1),
            (e.get("bbox") or [0, 0, 0, 0])[1],
        )
    )

    quality = _assess_quality(elements, selected_elements)
    metadata = _build_metadata(
        layout, target, quality,
        is_bilingual=True,
        structural_filter_used=True,
    )
    # Добавляем статистику билингвальности для диагностики
    metadata["language_filter"]["bilingual_blocks_found"] = bilingual_block_count

    return {
        "document_version_id": layout.get("document_version_id"),
        "document_id": layout.get("document_id"),
        "elements": selected_elements,
        "tables": tables,
        "metadata": metadata,
    }


# ---------------------------------------------------------------------------
# Главная точка входа
# ---------------------------------------------------------------------------

def filter_layout_to_language(
    layout: Dict[str, Any],
    target_language: str,
    context: Optional[Dict[str, Any]] = None,
) -> Dict[str, Any]:
    """
    Отфильтровать layout до одного языка.

    Аргументы:
      layout          — text_layout_raw (см. формат выше)
      target_language — "ru" или "en"
      context         — опциональный dict с ключами:
                        "is_bilingual" (bool) — флаг из extract_worker
                        "document_id"  (str)  — для логов
                        "deal_id"      (str)  — для логов

    Возвращает layout_primary (см. формат выше).
    """
    target = (target_language or "").lower()
    if target not in ("ru", "en"):
        # Неизвестный язык — возвращаем как есть, не ломаем пайплайн
        logger.warning(
            "filter_layout_to_language: unsupported target language, returning raw",
            target_language=target_language,
        )
        return layout

    elements: List[Dict[str, Any]] = layout.get("elements") or []
    tables: List[Any] = layout.get("tables") or []

    if not elements:
        return {
            "document_version_id": layout.get("document_version_id"),
            "document_id": layout.get("document_id"),
            "elements": [],
            "tables": tables,
            "metadata": _build_metadata(
                layout, target,
                _assess_quality([], []),
                is_bilingual=False,
                structural_filter_used=False,
            ),
        }

    is_bilingual = bool(context.get("is_bilingual")) if context else False
    doc_id = (context or {}).get("document_id", "unknown")

    # Для коротких документов или мономодальных — простой фильтр без анализа
    if not is_bilingual or len(elements) < _STRUCTURAL_MIN_ELEMENTS:
        return _simple_filter(layout, target, elements, tables, is_bilingual=is_bilingual)

    # Билингвальный режим — пробуем структурный анализ
    try:
        result = _structural_filter(layout, target, elements, tables)
        logger.info(
            "filter_layout_to_language: structural filter applied",
            document_id=doc_id,
            target=target,
            elements_in=len(elements),
            elements_out=len(result.get("elements") or []),
            bilingual_blocks=result.get("metadata", {})
                .get("language_filter", {})
                .get("bilingual_blocks_found", 0),
        )
        return result

    except Exception as exc:  # noqa: BLE001
        # Структурный анализ упал — fallback на простой фильтр.
        # Логируем полный traceback чтобы понять причину.
        logger.warning(
            "filter_layout_to_language: structural filter failed, "
            "falling back to simple filter",
            document_id=doc_id,
            target=target,
            error=str(exc),
            traceback=traceback.format_exc(),
        )
        return _simple_filter(layout, target, elements, tables, is_bilingual=True)


# ---------------------------------------------------------------------------
# Удобные алиасы
# ---------------------------------------------------------------------------

def filter_layout_to_ru(layout: Dict[str, Any]) -> Dict[str, Any]:
    return filter_layout_to_language(layout, "ru")


def filter_layout_to_en(layout: Dict[str, Any]) -> Dict[str, Any]:
    return filter_layout_to_language(layout, "en")

"""
Сервис для обновления языков сделки (Deal.languages и primary_language).

Назначение:
- аккуратно добавлять новые языки в список Deal.languages
  после появления честной информации (например, после извлечения layout);
- не затирать уже сохранённые языки, а объединять их с новыми;
- по возможности автоматически выставлять primary_language, если это безопасно.
"""

from __future__ import annotations

import uuid
from typing import Iterable

from sqlalchemy.ext.asyncio import AsyncSession

from db.repositories.deal_repo import DealRepository


async def update_deal_languages(
    db: AsyncSession,
    deal_id: uuid.UUID,
    new_languages: Iterable[str],
) -> None:
    """
    Обновить Deal.languages и при необходимости скорректировать primary_language.

    Для непрофессионального программиста:
    - deal.languages предполагается списком строк, например ["ru", "en"].
    - Функция делает три шага:
      1) читает из БД текущие языки сделки и текущий primary_language;
      2) объединяет их с new_languages (без дублей);
      3) сохраняет обновлённый список и, если это безопасно, обновляет primary_language.

    Поведение для primary_language:
    - если после объединения в списке ровно один язык → делаем его primary_language;
    - если языков несколько:
        * если среди них есть "ru" → primary_language остаётся как есть
          (предполагаем, что юрист сам выберет в UI);
        * если "ru" нет и primary_language пока пустой → берём первый язык
          из отсортированного списка (например, "en").
    - если сделка не найдена, функция просто выйдет без ошибок.
    """
    repo = DealRepository(db)
    deal = await repo.get_by_id(deal_id)
    if not deal:
        return

    # 1. Текущее состояние
    existing_languages = set(deal.languages or [])
    current_primary = deal.primary_language

    # 2. Добавляем новые языки
    for lang in new_languages:
        if not lang:
            continue
        existing_languages.add(lang)

    if not existing_languages:
        # Вообще нет языков — ничего не меняем.
        return

    # Приводим к отсортированному списку для предсказуемого порядка.
    languages_sorted = sorted(existing_languages)

    # 3. Вычисляем новое значение primary_language (если нужно).
    new_primary: str | None = current_primary

    if len(languages_sorted) == 1:
        # Ровно один язык — его логично считать основным.
        new_primary = languages_sorted[0]
    else:
        # Несколько языков.
        has_ru = "ru" in languages_sorted

        if not has_ru and not current_primary:
            # Русского нет, явного primary ещё не было — ставим первым
            # встречным (по алфавиту) языком, например "en".
            new_primary = languages_sorted[0]
        # Если есть "ru" или уже задан current_primary — не трогаем,
        # пусть выбор делает юрист через UI.

    # 4. Сохраняем изменения в БД через репозиторий.
    await repo.update_languages(
        deal_id=deal_id,
        languages=languages_sorted,
    )

    # Обновляем primary_language только если мы его реально изменили.
    if new_primary != current_primary:
        await repo.update_primary_language(
            deal_id=deal_id,
            primary_language=new_primary,
        )

"""
Repository for DocumentVersion CRUD operations.

Для непрофессионального программиста:

- Этот репозиторий — единственная "официальная точка", через которую стадии
  пайплайна должны обновлять ссылки на артефакты DocumentVersion.
- Здесь мы храним не только legacy refs (raw/sanitized), но и новые refs
  для одноязычного пайплайна:
  - layout_primary_ref;
  - layout_primary_sanitized_ref.
- Благодаря этому downstream-стадии (PII, structure, content service, checks)
  могут брать нужный layout из БД, а не вычислять путь в MinIO вручную.

Важно:
- Значение None в update_refs() означает "это поле не обновляем".
- То есть этот метод НЕ используется для очистки ссылок.
- Если когда-нибудь понадобится явно очищать ref, лучше сделать для этого
  отдельный метод, чтобы не было двусмысленности.
"""

import uuid

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from ..models.document import DocumentVersion


class DocumentVersionRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(
        self,
        document_id: uuid.UUID,
        version_num: int = 1,
        original_ref: str | None = None,
        origin_filetype: str | None = None,
        origin_extension: str | None = None,
    ) -> DocumentVersion:
        """Create a new document version."""
        version = DocumentVersion(
            document_id=document_id,
            version_num=version_num,
            original_ref=original_ref,
            origin_filetype=origin_filetype,
            origin_extension=origin_extension,
        )
        self.session.add(version)
        await self.session.flush()
        return version

    async def get_by_id(self, version_id: uuid.UUID) -> DocumentVersion | None:
        """Get document version by ID."""
        result = await self.session.execute(
            select(DocumentVersion).where(DocumentVersion.id == version_id)
        )
        return result.scalar_one_or_none()

    async def get_latest(self, document_id: uuid.UUID) -> DocumentVersion | None:
        """Get the latest version of a document."""
        result = await self.session.execute(
            select(DocumentVersion)
            .where(DocumentVersion.document_id == document_id)
            .order_by(DocumentVersion.version_num.desc())
            .limit(1)
        )
        return result.scalar_one_or_none()

    async def update_refs(
        self,
        version_id: uuid.UUID,
        raw_ref: str | None = None,
        layout_primary_ref: str | None = None,
        sanitized_ref: str | None = None,
        layout_primary_sanitized_ref: str | None = None,
        clause_tree_ref: str | None = None,
        extraction_strategy: str | None = None,
        ocr_confidence: float | None = None,
        pipeline_strategy: str | None = None,
        converted_pdf_ref: str | None = None,
    ) -> DocumentVersion | None:
        """
        Update artifact references and processing metadata on a document version.

        Используется разными стадиями:
        - convert_worker:
            converted_pdf_ref
        - extract_worker:
            raw_ref, layout_primary_ref, extraction_strategy,
            ocr_confidence, pipeline_strategy
        - pii_sanitizer:
            sanitized_ref, layout_primary_sanitized_ref
        - structure_builder:
            clause_tree_ref

        Важно:
        - raw_ref / sanitized_ref оставляем для backward compatibility;
        - layout_primary_ref / layout_primary_sanitized_ref — это новый основной
          контракт для одноязычного downstream-пайплайна;
        - None означает "не обновлять это поле";
        - если version_id не найден, метод возвращает None.
        """
        values: dict[str, str | float] = {}

        if raw_ref is not None:
            values["text_layout_raw_ref"] = raw_ref

        if layout_primary_ref is not None:
            values["layout_primary_ref"] = layout_primary_ref

        if sanitized_ref is not None:
            values["text_layout_sanitized_ref"] = sanitized_ref

        if layout_primary_sanitized_ref is not None:
            values["layout_primary_sanitized_ref"] = layout_primary_sanitized_ref

        if clause_tree_ref is not None:
            values["clause_tree_ref"] = clause_tree_ref

        if extraction_strategy is not None:
            values["extraction_strategy"] = extraction_strategy

        if ocr_confidence is not None:
            values["ocr_confidence"] = ocr_confidence

        if pipeline_strategy is not None:
            values["pipeline_strategy"] = pipeline_strategy

        if converted_pdf_ref is not None:
            values["converted_pdf_ref"] = converted_pdf_ref

        if not values:
            return await self.get_by_id(version_id)

        existing = await self.get_by_id(version_id)
        if not existing:
            return None

        stmt = (
            update(DocumentVersion)
            .where(DocumentVersion.id == version_id)
            .values(**values)
            .returning(DocumentVersion)
        )
        result = await self.session.execute(stmt)
        version = result.scalar_one_or_none()

        if version:
            await self.session.flush()

        return version

"""
Repository for Document CRUD and classification operations.

Этот модуль инкапсулирует всю работу с таблицей документов:

- создание записей о документах сделки;
- получение документов:
  - по id;
  - по сделке;
  - по SHA‑256 (для дедупликации);
- обновление статуса документа (pending, extracting, extracted, sanitized,
  structured, classified, completed, quarantine, failed и т.д.);
- обновление результатов классификации:
  - doc_type;
  - уровень и уверенность (classification_level / classification_confidence);
  - детали классификации;
  - applicable_triggers;
  - язык, юрисдикция;
  - версия playbook;
- вспомогательные выборки для аналитики:
  - количество по статусам;
  - выборка по типу и минимальной уверенности;
  - документы с низкой уверенностью.

Отдельно важен единый контракт по ошибкам обработки:

- processing_error_details — JSON‑поле в Document, куда мы пишем,
  что именно пошло не так по конкретному документу;
- методы set_failed / set_quarantined — единственные точки,
  через которые бизнес‑логика и воркеры должны помечать документ
  как "FAILED" или "QUARANTINE" и записывать детали ошибки.

Структура processing_error_details:

    {
        "stage": "<имя стадии пайплайна>",
        "type": "failed" | "quarantine",
        "reason": "<машинное короткое имя причины>" (опционально),
        "message": "<человекочитаемое описание ошибки>",
        "timestamp": "<ISO8601 UTC>",
        "run_id": "<UUID Run, если есть>" (опционально)
    }

Зачем это нужно юристу и фронтенду:

- по статусу документа и полю processing_error_details UI всегда может показать,
  на каком шаге документ остановился (import, extract, pii, structure, classify, checks, export),
  и понятное сообщение, что произошло;
- все стадии пайплайна (все воркеры) используют один и тот же формат ошибки,
  поэтому фронт не должен подстраиваться под разные схемы.
"""

import uuid

from sqlalchemy import func, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from core.constants import DocumentStatus
from ..models.document import Document


class DocumentRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(
        self,
        deal_id: uuid.UUID,
        bundle_id: uuid.UUID,
        filename: str,
        folder_path: str | None = None,
        sha256_file: str | None = None,
        status: str = "pending",
        source_extension: str | None = None,
        source_filetype: str | None = None,
    ) -> Document:
        """
        Создать новую запись документа в сделке.

        Для юриста это означает:
        - «в системе появился ещё один документ по сделке»;
        - документ пока может быть даже не обработан (status=pending).
        """
        doc = Document(
            deal_id=deal_id,
            bundle_id=bundle_id,
            filename=filename,

        self.session.add(doc)

        return doc

        Вернуть документы по сделке, опционально фильтруя по статусу и типу.

        stmt = select(Document).where(Document.deal_id == deal_id)

    async def update_status(self, doc_id: uuid.UUID, status: str, **kwargs) -> Document | None:
        """
        Обновить статус документа и дополнительные поля.

        Важно:
        - этот метод не навязывает формат processing_error_details;
        - для FAILED / QUARANTINE нужно использовать set_failed / set_quarantined,
          чтобы не потерять единый контракт по ошибкам.
        """
        values = {"status": status, **kwargs}
        stmt = (
            update(Document)
            .where(Document.id == doc_id)
            .values(**values)
            .returning(Document)
        )
        result = await self.session.execute(stmt)
        doc = result.scalar_one_or_none()
        if doc:
            await self.session.flush()
        return doc

    async def set_failed(
        self,
        doc_id: uuid.UUID,
        *,
        stage: str,
        message: str,
        reason: str | None = None,
        run_id: uuid.UUID | None = None,
    ) -> Document | None:
        """
        Перевести документ в FAILED и записать детали ошибки в processing_error_details.

        Использовать этот метод должны все стадии пайплайна (import, extract, pii,
        structure, classify, checks, export), когда ошибка по документу:
        - фатальна для этого документа;
        - и мы больше не можем продолжать обработку (например, битый PDF, нет clause_tree).

        Поле stage должно отражать стадию пайплайна, на которой остановился документ:
        - "import", "extract", "pii", "structure", "classify", "checks", "export" и т.п.
        """
        from datetime import datetime, UTC

        details: dict[str, object] = {
            "stage": stage,
            "type": "failed",
            "message": message,
            "timestamp": datetime.now(UTC).isoformat(),
        }
        if reason is not None:
            # reason — короткий машинный код причины (например, "pdf_corrupted").
            details["reason"] = reason
        if run_id is not None:
            # run_id — чтобы фронт мог показать, в рамках какого Run произошла ошибка.
            details["run_id"] = str(run_id)

        return await self.update_status(
            doc_id,
            DocumentStatus.FAILED.value,
            processing_error_details=details,
        )

    async def set_quarantined(
        self,
        doc_id: uuid.UUID,
        *,
        stage: str,
        message: str,
        reason: str | None = None,
        run_id: uuid.UUID | None = None,
    ) -> Document | None:
        """
        Перевести документ в QUARANTINE и записать детали ошибки.

        Использовать, когда:
        - документ формально обработан, но есть сомнения в корректности результата;
        - требуется ручная проверка юристом (например, очень плохое качество OCR,
          подозрение на неправильную классификацию и т.п.).

        Для юриста это выглядит как отдельный статус "quarantine" с пояснением,
        что именно вызывает сомнение.
        """
        from datetime import datetime, UTC

        details: dict[str, object] = {
            "stage": stage,
            "type": "quarantine",
            "message": message,
            "timestamp": datetime.now(UTC).isoformat(),
        }
        if reason is not None:
            details["reason"] = reason
        if run_id is not None:
            details["run_id"] = str(run_id)

        return await self.update_status(
            doc_id,
            DocumentStatus.QUARANTINE.value,
            processing_error_details=details,
        )

    async def set_completed(self, doc_id: uuid.UUID) -> Document | None:
        """
        Перевести документ в статус COMPLETED (полный анализ выполнен).

        Важно:
        - этот метод используется, когда по документу успешно прошли все стадии
          пайплайна, включая проверки по плейбуку (checks).
        - в отличие от FAILED / QUARANTINE, здесь мы не пишем processing_error_details,
          потому что это "нормальное" успешное завершение обработки.
        """
        return await self.update_status(
            doc_id,
            DocumentStatus.COMPLETED.value,
        )


    async def update_classification(
        self,
        doc_id: uuid.UUID,
        doc_type: str | None,
        classification_confidence: float,
        classification_level: str | None,
        classification_details: dict,
        applicable_triggers: list[str],
        classification_requires_review: bool | None = None,
        language: str | None = None,
        jurisdiction: str | None = None,
        playbook_version_id: uuid.UUID | None = None,
    ) -> Document | None:
        """
        Обновить результаты классификации документа.

        Это вызывается после успешной работы классификатора:
        - сохранение типа договора (doc_type);
        - уровня (L0–L4) и уверенности;
        - детальной информации о том, почему выбрана такая классификация;
        - языковых и юрисдикционных метаданных;
        - ID версии плейбука, по которому классифицировали документ.
        - classification_requires_review:
            флаг, который приходит из HybridClassifier.details["requires_manual_review"]
            и сохраняется в Document.classification_requires_review для UX.
        """
        values = {
            "doc_type": doc_type,
            "classification_confidence": float(classification_confidence),
            "classification_level": classification_level,
            "classification_details": classification_details,
            "applicable_triggers": applicable_triggers,
        }

        if classification_requires_review is not None:
            values["classification_requires_review"] = classification_requires_review
            
        # Обновляем язык и юрисдикцию, если они переданы.
        if language:
            values["language"] = language
        if jurisdiction:
            values["jurisdiction_guess"] = jurisdiction

        if playbook_version_id is not None:
            values["playbook_version_id"] = playbook_version_id

        stmt = (
            update(Document)
            .where(Document.id == doc_id)
            .values(**values)
            .returning(Document)
        )
        result = await self.session.execute(stmt)
        doc = result.scalar_one_or_none()
        if doc:
            await self.session.flush()
        return doc

    async def count_by_status(self, deal_id: uuid.UUID) -> dict[str, int]:
        """
        Посчитать количество документов по статусам для сделки.

        Фронт может использовать это для сводки:
        - сколько документов в работе;
        - сколько уже полностью обработаны;

            .group_by(Document.status)

    async def get_page_for_deal(
        self,
        deal_id: uuid.UUID,
        page: int,
        per_page: int,
        include_classification: bool = False,
    ) -> tuple[list[Document], int]:
        """
        Вернуть страницу документов сделки + общее количество.

        Это основа для пагинации в UI:
        - список документов по страницам;
        - возможность сортировать по статусу/типу/уверенности.
        """

        # 1) total
        count_stmt = select(func.count(Document.id)).where(Document.deal_id == deal_id)
        total_result = await self.session.execute(count_stmt)
        total = total_result.scalar_one() or 0

        if total == 0:
            return [], 0

        stmt = select(Document).where(Document.deal_id == deal_id)

        if include_classification:
            stmt = stmt.order_by(
                Document.status,
                Document.doc_type.is_(None),
                Document.doc_type,
                Document.classification_confidence.desc().nullslast(),
                Document.folder_path,
                Document.filename,
            )
        else:
            stmt = stmt.order_by(
                Document.folder_path,
                Document.filename,
            )

        stmt = stmt.offset(offset).limit(per_page)
        result = await self.session.execute(stmt)
        items = list(result.scalars().all())

        return items, total

        return count > 0

"""
Репозиторий для CRUD‑операций со сделками (Deal).

Это тонкая обёртка над SQLAlchemy ORM:
- никакой бизнес‑логики;
- только чтение/запись/поиск сделок в БД;
- все операции выполняются через переданную AsyncSession.
"""

import uuid

from sqlalchemy import select, update, func
from sqlalchemy.ext.asyncio import AsyncSession

from ..models.deal import Deal


class DealRepository:
    """
    Класс-репозиторий для работы со сделками.

    Экземпляр живёт внутри одного запроса / юнита работы и использует
    уже созданную AsyncSession — сам сессию не открывает и не закрывает.
    """

    def __init__(self, session: AsyncSession):
        """
        Инициализировать репозиторий.

        Параметры:
        - session: активная AsyncSession, через которую будут выполняться запросы.
        """
        self.session = session

    async def create(
        self,
        tenant_id: uuid.UUID,
        name: str,
        deal_type: str | None = None,
        jurisdiction: str | None = None,
        languages: list[str] | None = None,
        pii_mode: str = "local_only",
        profile: str = "full",
        playbook_version_id: uuid.UUID | None = None,
        metadata_: dict | None = None,
    ) -> Deal:
        """
        Создать новую сделку и сохранить её в БД.

        Параметры:
        - tenant_id: идентификатор арендатора, к которому относится сделка;
        - name: человекочитаемое название сделки;
        - deal_type: тип сделки (опционально, для фильтрации и плейбуков);
        - jurisdiction: юрисдикция сделки;
        - languages: список языков документов по сделке;
        - pii_mode: режим работы с PII (по умолчанию 'local_only');
        - profile: профиль анализа (например, 'full'/'light');
        - playbook_version_id: ID версии плейбука, к которой привязана сделка;
        - metadata_: произвольный JSON с дополнительными метаданными.

        Возвращает:
        - созданный объект Deal (уже с заполненным id).
        """
        deal = Deal(
            tenant_id=tenant_id,
            name=name,
            deal_type=deal_type,
            jurisdiction=jurisdiction,
            languages=languages,
            pii_mode=pii_mode,
            profile=profile,
            playbook_version_id=playbook_version_id,
            metadata_=metadata_ or {},
        )
        self.session.add(deal)
        await self.session.flush()
        return deal

    async def get_by_id(self, deal_id: uuid.UUID) -> Deal | None:
        """
        Найти сделку по её ID.

        Параметры:
        - deal_id: идентификатор сделки.

        Возвращает:
        - объект Deal, если найден;
        - None, если сделки с таким ID нет.
        """
        result = await self.session.execute(select(Deal).where(Deal.id == deal_id))
        return result.scalar_one_or_none()

    async def get_by_tenant(
        self,
        tenant_id: uuid.UUID,
        status: str | None = None,
    ) -> list[Deal]:
        """
        Вернуть список сделок для указанного арендатора.

        Параметры:
        - tenant_id: идентификатор арендатора;
        - status: опциональный фильтр по статусу сделки.

        Возвращает:
        - список объектов Deal, отсортированный по дате создания (новые сверху).
        """
        stmt = select(Deal).where(Deal.tenant_id == tenant_id)
        if status is not None:
            stmt = stmt.where(Deal.status == status)
        stmt = stmt.order_by(Deal.created_at.desc())
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def get_all(self) -> list[Deal]:
        """
        Вернуть все сделки в системе.

        Используется в dev‑режиме и админских сценариях, где нужно увидеть
        полный список сделок без фильтров.
        """
        stmt = select(Deal).order_by(Deal.created_at.desc())
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def update_status(self, deal_id: uuid.UUID, status: str) -> Deal | None:
        """
        Обновить статус сделки.

        Параметры:
        - deal_id: идентификатор сделки;
        - status: новый статус (строковое значение, например 'active', 'archived').

        Возвращает:
        - обновлённый объект Deal;
        - None, если сделки с таким ID не найдено.
        """
        stmt = (
            update(Deal)
            .where(Deal.id == deal_id)
            .values(status=status)
            .returning(Deal)
        )
        result = await self.session.execute(stmt)
        deal = result.scalar_one_or_none()
        if deal:
            await self.session.flush()
        return deal

    async def get_page(
        self,
        page: int,
        per_page: int,
        status: str | None = None,
        tenant_id: uuid.UUID | None = None,
    ) -> tuple[list[Deal], int]:
        """
        Вернуть одну страницу сделок с учётом фильтров и общее количество записей.

        Параметры:
        - page: номер страницы (начиная с 1);
        - per_page: размер страницы (количество сделок на странице);
        - status: опциональный фильтр по статусу сделки;
        - tenant_id: опциональный фильтр по арендатору.

        Возвращает кортеж:
        - список сделок на запрошенной странице;
        - общее количество сделок, подходящих под фильтры (для расчёта пагинации).
        """
        offset = (page - 1) * per_page

        filters = []
        if status is not None:
            filters.append(Deal.status == status)
        if tenant_id is not None:
            filters.append(Deal.tenant_id == tenant_id)

        # 1) считаем total с учётом фильтров
        count_stmt = select(func.count(Deal.id))
        if filters:
            count_stmt = count_stmt.where(*filters)

        total_result = await self.session.execute(count_stmt)
        total = total_result.scalar_one() or 0

        if total == 0:
            return [], 0

        # 2) выбираем записи страницы
        stmt = (
            select(Deal)
            .where(*filters)
            .order_by(Deal.created_at.desc())
            .offset(offset)
            .limit(per_page)
        )
        result = await self.session.execute(stmt)
        items = list(result.scalars().all())

        return items, total

    async def set_last_run_id(
        self,
        deal_id: uuid.UUID,
        run_id: uuid.UUID | None,
        checks_mode: str | None = None,
        pipeline_strategy: str | None = None,
    ) -> None:
        """
        Обновить у сделки ссылку на последний Run и его параметры.

        Обновляет в таблице Deal:
        - last_run_id                 — идентификатор последнего Run;
        - last_run_checks_mode        — режим проверок последнего Run (если передан);
        - last_run_pipeline_strategy  — стратегию пайплайна последнего Run (если передана).

        Эти поля используются в UI и отчётах, чтобы быстро показать,
        в каком режиме и с какой стратегией был выполнен последний анализ сделки.
        """
        values: dict[str, object] = {"last_run_id": run_id}
        if checks_mode is not None:
            values["last_run_checks_mode"] = checks_mode
        if pipeline_strategy is not None:
            values["last_run_pipeline_strategy"] = pipeline_strategy

        stmt = (
            update(Deal)
            .where(Deal.id == deal_id)
            .values(**values)

"""
DocumentContentService v3: тонкий фасад над clause tree + layout.

Простыми словами:

- В MinIO есть два типа артефактов:
  1) layout (text layout) — список маленьких текстовых кусочков (elements);

class ClauseTreeStorageClientProtocol:

    Для непрофессионального программиста:

class ClauseTreeStorageClient(ClauseTreeStorageClientProtocol):
    """
    Адаптер поверх ArtifactStore, реализующий load_json(uri).

    ArtifactStore сам знает, как читать JSON из MinIO.
    Этот класс просто оборачивает синхронный вызов в асинхронный,
    чтобы удобнее использовать его в async-коде.
    """

    def __init__(self, store: ArtifactStore | None = None) -> None:
        self.store = store or ArtifactStore()

    async def load_json(self, uri: str) -> dict[str, Any]:
        from anyio.to_thread import run_sync

        def _get() -> dict[str, Any]:
            return self.store.get_json(uri)

        return await run_sync(_get)

class DocumentContentService:
    """
    Сервис доступа к содержимому документа (структура + текст).

    Важно:

    - Сервис работает как "адаптер" поверх:
        * AsyncSession (через DocumentVersionRepository),
        * ArtifactStore (через ClauseTreeStorageClient и LayoutStorageClient).
    - Вся сложная логика вынесена в отдельные модули:
        * ClauseTreeBuilder — обход clause tree;
        * chunking.* — сплит текста и иерархия чанков.
    - Никаких глобальных синглтонов и кэшей здесь нет.
      Каждый воркер создаёт этот сервис в рамках одного UnitOfWork (db_uow).
    """

    def __init__(
        self,
        session: AsyncSession,
        storage_client: ClauseTreeStorageClientProtocol | None = None,
        layout_client: LayoutStorageClient | None = None,
    ) -> None:
        self.session = session
        self.versions = DocumentVersionRepository(session)
        self.storage = storage_client or ClauseTreeStorageClient()
        self.layout_storage = layout_client or LayoutStorageClient()

    # -------------------------------------------------------------------------
    # Внутренние helpers: загрузка последней версии, clause tree и layout
    # -------------------------------------------------------------------------

    async def _get_latest_version(self, doc_id: uuid.UUID) -> Optional[DocumentVersion]:
        """
        Получить последнюю версию документа (по version_num).

        Если версий нет, вернёт None.
        """
        return await self.versions.get_latest(doc_id)

    async def _load_clause_tree(self, version: DocumentVersion) -> Optional[Dict[str, Any]]:
        """
        Загрузить clause tree JSON по DocumentVersion.

        Если ссылки нет — вернёт None.
        """
        if not version.clause_tree_ref:
            return None
        return await self.storage.load_json(version.clause_tree_ref)

    async def _load_layout(self, version: DocumentVersion) -> Optional[Dict[str, Any]]:
        """
        Загрузить рабочий layout документа по единому контракту пайплайна.

        Приоритет:
        1) layout_primary_sanitized_ref — основной source of truth после PII;
        2) layout_primary_ref — recovery fallback для случаев, когда PII ещё не выполнен.
        """
        layout_ref = (
            getattr(version, "layout_primary_sanitized_ref", None)
            or getattr(version, "layout_primary_ref", None)
        )
        if not layout_ref:
            return None
        return await self.layout_storage.load_json(layout_ref)

    # -------------------------------------------------------------------------
    # Вспомогательный расчёт position_ratio
    # -------------------------------------------------------------------------

    def _calc_position_ratio(
        self,
        start_page: int | None,
        end_page: int | None,
        total_pages: int | None,
    ) -> float | None:
        """
        Посчитать нормализованную позицию чанка в документе.

        Для непрофессионального программиста:
        - Если знаем страницу(ы) чанка и общее число страниц,
          считаем "середину" чанка и делим на длину документа.
        - Получаем число от 0.0 до 1.0:
            ~0.0  — начало документа,
            ~0.5  — середина,
            ~1.0  — конец.
        """
        if not total_pages or total_pages <= 0:
            return None

        if start_page is None and end_page is None:
            return None

        s = start_page if isinstance(start_page, int) else end_page
        e = end_page if isinstance(end_page, int) else start_page
        if s is None or e is None:
            return None

        mid = (s + e) / 2.0
        ratio = mid / float(total_pages)
        if ratio < 0.0:
            return 0.0
        if ratio > 1.0:
            return 1.0
        return ratio

    # -------------------------------------------------------------------------
    # Публичные методы
    # -------------------------------------------------------------------------

    async def get_title_and_preamble(
        self,
        doc_id: uuid.UUID,
        max_pages: int = 2,
        max_length: int = 1500,
    ) -> Optional[str]:
        """
        Извлечь шапку документа (title + преамбула) напрямую из layout.

        Простыми словами:
        - Берём layout последней версии документа.
        - Сканируем первые несколько страниц (обычно 1–2).
        - Собираем текст сверху вниз, пока не встретим первое крупное
          структурное разделение (оглавление или раздел уровня "1. DEFINITIONS").
        - Возвращаем один текстовый блок для L2/L4 и фронта.
        """
        version = await self._get_latest_version(doc_id)
        if not version:
            return None

        layout = await self._load_layout(version)
        if not layout:
            return None

        elements = layout.get("elements") or []
        if not elements:
            return None

        # Фильтруем только первые max_pages страниц
        header_elements: List[Dict[str, Any]] = []
        for el in elements:
            page = el.get("page")
            if not isinstance(page, int) or page > max_pages:
                continue

            text = (el.get("text") or "").strip()
            if not text:
                continue

            upper = text.upper()

            # Стоп‑сигналы: оглавление или явный крупный раздел
            if "TABLE OF CONTENTS" in upper or "СОДЕРЖАНИЕ" in upper:
                break

            # Примитивная эвристика: если строка похожа на раздел 1./1.1 и т.п.,
            # считаем, что шапка закончилась.
            import re

            if re.match(r"^\d{1,2}(\.\d{1,2})*\.\s+.+", text):
                break

            header_elements.append(el)

        if not header_elements:
            return None

        parts: List[str] = []
        for el in header_elements:
            t = (el.get("text") or "").strip()
            if t:
                parts.append(t)

        full = "\n".join(parts).strip()
        if not full:
            return None

        if len(full) > max_length:
            full = full[:max_length]

        return full or None

    async def get_chunks(self, doc_id: uuid.UUID) -> List[Dict[str, Any]]:
        """
        Построить чанки для Retrieval/Checks:

        - один чанк на каждую листовую клаузу в clause tree;
        - текст чанка — реальный текст клаузы, собранный по element_ids
          из layout (sanitized/raw);
        - если clause tree отсутствует или layout недоступен, вернуть один
          "пустой" чанк, чтобы пайплайн не падал.

        Для непрофессионального программиста:
        - "чанк" = кусок текста договора, обычно один пункт,
          который дальше анализируют на риски.
        """
        version = await self._get_latest_version(doc_id)
        if not version:
            logger.warning(
                "DocumentContentService.get_chunks: no latest version, returning empty fallback chunk",
                extra={"doc_id": str(doc_id)},
            )
            return [
                {
                    "id": f"{doc_id}:full",
                    "text": "",
                    "metadata": {"doc_id": str(doc_id)},
                }
            ]

        tree = await self._load_clause_tree(version)
        layout = await self._load_layout(version)

        if not tree or not layout:
            logger.warning(
                "DocumentContentService.get_chunks: missing clause tree or layout, returning empty fallback chunk",
                extra={
                    "doc_id": str(doc_id),
                    "has_tree": bool(tree),
                    "has_layout": bool(layout),
                    "clause_tree_ref": getattr(version, "clause_tree_ref", None),
                    "layout_primary_ref": getattr(version, "layout_primary_ref", None),
                    "layout_primary_sanitized_ref": getattr(version, "layout_primary_sanitized_ref", None),
                },
            )
            return [
                {
                    "id": f"{doc_id}:full",
                    "text": "",
                    "metadata": {"doc_id": str(doc_id)},
                }
            ]

        leaf_chunks = ClauseTreeBuilder.build_leaf_chunks(
            doc_id=doc_id,
            tree=tree,
            layout=layout,
        )

        from core.logging import get_logger
        logger = get_logger(__name__)
        if str(doc_id) == "7012e86f-89a2-46b3-ac1a-55afa3c989c4":
            logger.info(
                "DEBUG chunk 8 sample",
                layout_ref=getattr(version, "layout_primary_sanitized_ref", None)
                        or getattr(version, "layout_primary_ref", None)
                        or getattr(version, "text_layout_sanitized_ref", None)
                        or getattr(version, "text_layout_raw_ref", None),
                chunk_8_text=(leaf_chunks[7].get("text") if len(leaf_chunks) > 7 else None),
            )

        if not leaf_chunks:
            logger.warning(
                "DocumentContentService.get_chunks: clause tree produced no leaf chunks, returning empty fallback chunk",
                extra={
                    "doc_id": str(doc_id),
                    "clause_tree_ref": getattr(version, "clause_tree_ref", None),
                    "layout_primary_ref": getattr(version, "layout_primary_ref", None),
                    "layout_primary_sanitized_ref": getattr(version, "layout_primary_sanitized_ref", None),
                },
            )
            return [
                {
                    "id": f"{doc_id}:full",
                    "text": "",
                    "metadata": {"doc_id": str(doc_id)},
                }
            ]

        # Пытаемся достать общее число страниц из layout.metadata
        total_pages: int | None = None
        layout_meta = layout.get("metadata") or {}
        # возможные ключи: page_count, pages и т.п.
        for key in ("page_count", "pages", "num_pages", "total_pages"):
            value = layout_meta.get(key)
            if isinstance(value, int) and value > 0:
                total_pages = value
                break

        # Пробрасываем информацию о language_filter / degraded из layout в метаданные чанков.
        lf_meta = (layout_meta.get("language_filter") or {}) if isinstance(layout_meta, dict) else {}
        lf_degraded = lf_meta.get("degraded")
        lf_reason = lf_meta.get("degraded_reason")
        lf_target_lang = lf_meta.get("target_language")

        if lf_degraded is not None or lf_target_lang:
            for ch in leaf_chunks:
                meta = ch.get("metadata") or {}
                lf = dict(meta.get("language_filter") or {})
                if lf_target_lang is not None and "target_language" not in lf:
                    lf["target_language"] = lf_target_lang
                if lf_degraded is not None and "degraded" not in lf:
                    lf["degraded"] = bool(lf_degraded)
                if lf_reason is not None and "degraded_reason" not in lf:
                    lf["degraded_reason"] = lf_reason
                if lf:
                    meta["language_filter"] = lf
                    ch["metadata"] = meta

        # Добавляем position_ratio для каждого листового чанка (если можем)
        if total_pages:
            for ch in leaf_chunks:
                meta = ch.get("metadata") or {}
                start_page = meta.get("start_page")
                end_page = meta.get("end_page")
                pos = self._calc_position_ratio(start_page, end_page, total_pages)
                if pos is not None:
                    meta["position_ratio"] = pos
                    ch["metadata"] = meta

        return leaf_chunks

    async def get_header_and_snippet(
        self,
        doc_id: uuid.UUID,
        max_snippet_len: int = 1500,
        max_leaf_clauses: int = 3,
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Вернуть:

        - header_text: заголовки верхнего уровня (обычно level 1) —
          для L2 классификатора;
        - snippet_text: текст первых нескольких листовых клауз —
          для L3 (эмбеддинги) и L4 (LLM).

        Важно:
        - snippet собирается по тем же правилам, что и чанки:
          по element_ids из clause tree, через layout.
        """
        version = await self._get_latest_version(doc_id)
        if not version:
            return None, None

        tree = await self._load_clause_tree(version)
        layout = await self._load_layout(version)

        if not tree or not layout:
            return None, None

        # Header: нормализованная шапка (title + преамбула) из layout.
        header_text = await self.get_title_and_preamble(doc_id)

        # Листовые клаузы — используем тот же ClauseTreeBuilder.
        leaf_chunks = ClauseTreeBuilder.build_leaf_chunks(
            doc_id=doc_id,
            tree=tree,
            layout=layout,
        )

        if not leaf_chunks:
            return header_text, None

        # Берём первые max_leaf_clauses клауз.
        leaf_chunks = leaf_chunks[:max_leaf_clauses]

        # --- Микрополитика длины сниппета ---
        MIN_CLAUSE_LEN = 80      # минимум символов для "осмысленной" клаузы
        EXTRA_CLAUSES = 5        # сколько максимум клауз можно взять в сниппет

        # Посчитаем суммарную длину первых двух клауз
        first_two = leaf_chunks[:2]
        first_two_len = sum(len(ch.get("text") or "") for ch in first_two)

        # Если короткие — расширяем окно до EXTRA_CLAUSES (если они есть)
        if first_two_len < MIN_CLAUSE_LEN:
            extra_leaf_chunks = leaf_chunks
            if len(extra_leaf_chunks) < EXTRA_CLAUSES:
                # На всякий случай можно было бы добавить клаузы из дерева,
                # но сейчас у нас уже есть первые leaf_chunks.
                pass
            leaf_chunks = extra_leaf_chunks[:EXTRA_CLAUSES]

        # Собираем текст сниппета: клаузы разделяем двойным переводом строки,
        # чтобы модель видела границы пунктов.
        texts_for_snippet: List[str] = []
        for ch in leaf_chunks:
            text = (ch.get("text") or "").strip()
            if text:
                texts_for_snippet.append(text)

        snippet_text = "\n\n".join(texts_for_snippet).strip()

        if snippet_text and len(snippet_text) > max_snippet_len:
            snippet_text = snippet_text[:max_snippet_len]

        return header_text, (snippet_text or None)

    async def _get_clause_chunks_as_retrieval(
        self,
        doc_id: uuid.UUID,
    ) -> list[RetrievalChunk]:
        """
        Внутренний адаптер: get_chunks -> список RetrievalChunk с kind="clause".

        Для непрофессионального программиста:
        - Мы берём уже готовые клауза-чанки (dict) из get_chunks();
        - Превращаем каждый в объект RetrievalChunk с нормальным "паспортом";
        - В дальнейшем retrieval / LLM / checks будут работать именно с этим типом.
        """
        raw_chunks = await self.get_chunks(doc_id)

        retrieval_chunks: list[RetrievalChunk] = []
        for ch in raw_chunks:
            text = (ch.get("text") or "").strip()
            meta = ch.get("metadata") or {}
            if not text:
                continue

            rc = RetrievalChunk(
                id=str(ch.get("id") or f"{doc_id}:clause"),
                doc_id=meta.get("doc_id") or str(doc_id),
                kind="clause",
                clause_uid=meta.get("clause_uid"),
                level=meta.get("level"),
                text=text,
                token_count=meta.get("token_count") or count_tokens(text),
                path_titles=list(meta.get("path_titles") or []),
                start_page=meta.get("start_page"),
                end_page=meta.get("end_page"),
                element_ids=[str(e) for e in meta.get("element_ids") or []],
                summary=meta.get("summary"),
                meta={
                    k: v
                    for k, v in meta.items()
                    if k
                    not in {
                        "doc_id",
                        "clause_uid",
                        "level",
                        "path_titles",
                        "start_page",
                        "end_page",
                        "element_ids",
                        "token_count",
                        "summary",
                    }
                },
            )
            retrieval_chunks.append(rc)

        return retrieval_chunks

    async def get_sized_chunks(
        self,
        doc_id: uuid.UUID,
        target_tokens: int = 300,
        max_tokens: int = 800,
        overlap_ratio: float = 0.15,
    ) -> List[Dict[str, Any]]:
        """
        Построить НОРМАЛИЗОВАННЫЕ чанки поверх листовых клауз.

        Для непрофессионального пользователя:
        - сначала каждая листовая клауза превращается в один крупный чанк (get_chunks);
        - затем мы берём эти большие куски и аккуратно делим их
          на более мелкие части по предложениям и абзацам;
        - цель: чтобы каждый чанк был удобного размера для нейросети,
          примерно около target_tokens, но не больше max_tokens;
        - небольшое перекрытие (overlap_ratio) нужно, чтобы смысл не терялся
          на стыках чанков: часть текста повторяется в соседнем чанке.

        На выходе:
        - возвращается список словарей с полями:
          * "id" — уникальный идентификатор чанка,
          * "text" — текст чанка,
          * "metadata" — служебные данные (пункт договора, страницы, путь в структуре и т.п.).
        """

        # Политика длины клаузы:
        # - короткие клаузы (<= SHORT_CLAUSE_MAX_TOKENS) не режем вообще;
        # - очень длинные (>= LONG_CLAUSE_MIN_TOKENS) режем;
        # - средние оставляем целиком, чтобы не дробить нормальные пункты.
        SHORT_CLAUSE_MAX_TOKENS = int(target_tokens * 0.9)   # ~270 при target=300
        LONG_CLAUSE_MIN_TOKENS = int(max_tokens * 1.2)       # ~960 при max=800

        base_chunks = await self.get_chunks(doc_id)
        sized: List[Dict[str, Any]] = []

        # Пытаемся вычислить total_pages ещё раз, если он есть в metadata базовых чанков.
        # (get_chunks уже пытался, но здесь мы подстраховываемся от будущих изменений.)
        total_pages: int | None = None
        for base in base_chunks:
            meta = base.get("metadata") or {}
            pos = meta.get("position_ratio")
            # если position_ratio уже есть, значит total_pages уже учитывался
            if pos is not None:
                total_pages = None
                break

        # Если вдруг потребуется, можно будет снова достать layout и пересчитать,
        # но сейчас считаем, что get_chunks уже сделал основную работу.

        for base in base_chunks:
            text = base.get("text") or ""
            metadata = base.get("metadata") or {}

            # Если текста нет — пропускаем такой чанк.
            if not text.strip():
                continue

            clause_uid = metadata.get("clause_uid")
            level = metadata.get("level")
            path_titles = metadata.get("path_titles") or []
            start_page = metadata.get("start_page")
            end_page = metadata.get("end_page")
            element_ids = metadata.get("element_ids") or []

            # Считаем длину клаузы в токенах один раз.
            clause_token_count = count_tokens(text)

            # Считаем position_ratio для всей клаузы (если ещё нет).
            if "position_ratio" in metadata:
                position_ratio = metadata.get("position_ratio")
            elif total_pages:
                position_ratio = self._calc_position_ratio(start_page, end_page, total_pages)
            else:
                position_ratio = None

            def _append_whole_clause_chunk() -> None:
                chunk_id = f"{doc_id}:{clause_uid or 'full'}:0"
                sized.append(
                    {
                        "id": chunk_id,
                        "text": text,
                        "metadata": {
                            "doc_id": str(doc_id),
                            "clause_uid": clause_uid,
                            "title": metadata.get("title"),
                            "level": level,
                            "path_titles": path_titles,
                            "start_page": start_page,
                            "end_page": end_page,
                            "element_ids": element_ids,
                            "token_count": clause_token_count,
                            "position_ratio": position_ratio,
                        },
                    }
                )

            # 1) Короткие клаузы: вообще не режем.
            if clause_token_count <= SHORT_CLAUSE_MAX_TOKENS:
                _append_whole_clause_chunk()
                continue

            # 2) Средние клаузы: по умолчанию тоже не режем,
            # чтобы сохранить одну «юридическую мысль» целиком.
            if SHORT_CLAUSE_MAX_TOKENS < clause_token_count < LONG_CLAUSE_MIN_TOKENS:
                _append_whole_clause_chunk()
                continue

            # 3) Только очень длинные клаузы режем на под-чанки.
            sub_texts = split_text_to_sized_chunks(
                text=text,
                target_tokens=target_tokens,
                max_tokens=max_tokens,
                overlap_ratio=overlap_ratio,
            )

            # Если по какой-то причине сплит не сработал — сохраняем исходный чанк.
            if not sub_texts:
                _append_whole_clause_chunk()
                continue

            # Собираем под-чанки с отдельными ID, но с той же метадатой.
            for sub_idx, sub_text in enumerate(sub_texts):
                if not sub_text.strip():
                    continue

                sub_token_count = count_tokens(sub_text)
                chunk_id = f"{doc_id}:{clause_uid or 'full'}:{sub_idx}"

                # Для под-чанков логично наследовать position_ratio родительской клаузы.
                sub_metadata = {
                    "doc_id": str(doc_id),
                    "clause_uid": clause_uid,
                    "title": metadata.get("title"),
                    "level": level,
                    "path_titles": path_titles,
                    "start_page": start_page,
                    "end_page": end_page,
                    "element_ids": element_ids,
                    "token_count": sub_token_count,
                    "position_ratio": position_ratio,
                }

                sized.append(
                    {
                        "id": chunk_id,
                        "text": sub_text,
                        "metadata": sub_metadata,
                    }
                )

        if not sized:
            logger.warning(
                "DocumentContentService.get_sized_chunks: no sized chunks built, returning empty fallback chunk",
                extra={"doc_id": str(doc_id)},
            )
            return [
                {
                    "id": f"{doc_id}:full",
                    "text": "",
                    "metadata": {"doc_id": str(doc_id)},
                }
            ]

        return sized

    async def get_hierarchical_chunks(
        self,
        doc_id: uuid.UUID,
        target_tokens: int = 300,
        max_tokens: int = 800,
        overlap_ratio: float = 0.15,
    ) -> Dict[str, List[Dict[str, Any]]]:
        """
        Вернуть многоуровневые чанки по документу.

        Для непрофессионального программиста:
        - section: крупные разделы договора (объединённые клаузы по заголовкам).
        - clause: исходные листовые клаузы (как в get_chunks()).
        - subclause: нормализованные под-чанки внутри клауз (как в get_sized_chunks()).

        ВАЖНО:
        - Внешний формат остаётся таким же, как был раньше:
            {
                "section": [ {id, text, metadata}, ... ],
                "clause":  [ {id, text, metadata}, ... ],
                "subclause": [ {id, text, metadata}, ... ],
            }
        - Внутри мы уже работаем с RetrievalChunk, чтобы иметь единый "паспорт"
          для всех уровней.
        """
        # 1. Берём clause/subclause как RetrievalChunk
        clause_chunks_rc = await self._get_clause_chunks_as_retrieval(doc_id)
        sub_chunks_rc = await self._get_sized_chunks_as_retrieval(
            doc_id=doc_id,
            target_tokens=target_tokens,
            max_tokens=max_tokens,
            overlap_ratio=overlap_ratio,
        )

        # 2. Для построения section-чанков пока используем старую функцию,
        #    которая ожидает dict-формат clause-чанков.
        clause_chunks_legacy = [rc.to_legacy_dict() for rc in clause_chunks_rc]
        section_chunks_legacy = build_section_chunks_from_clauses(
            doc_id=str(doc_id),
            clause_chunks=clause_chunks_legacy,
            max_tokens=1200,
        )

        # 3. Добавляем summary для clause/subclause поверх RetrievalChunk
        def _with_summary_rc(chunks: List[RetrievalChunk]) -> List[Dict[str, Any]]:
            out: List[Dict[str, Any]] = []
            for rc in chunks:
                if not rc.summary:
                    rc.summary = build_chunk_summary(rc.path_titles, rc.text)
                out.append(rc.to_legacy_dict())
            return out

        # 4. Возвращаем в старом формате
        return {
            "section": section_chunks_legacy,
            "clause": _with_summary_rc(clause_chunks_rc),
            "subclause": _with_summary_rc(sub_chunks_rc),
        }

    async def get_retrieval_chunks_for_checks(
        self,
        doc_id: uuid.UUID,
        levels: list[str] | None = None,
        target_tokens: int = 300,
        max_tokens: int = 800,
        overlap_ratio: float = 0.15,
    ) -> list[dict[str, Any]]:
        """
        Удобный фасад для CheckRunner:
        вернуть плоский список чанков для checks с mix уровней.

        По умолчанию:
        - subclause: нормализованные под-чанки,
        - section: крупные разделы.
        """
        if levels is None:
            levels = ["subclause", "section"]

        hier = await self.get_hierarchical_chunks(
            doc_id=doc_id,
            target_tokens=target_tokens,
            max_tokens=max_tokens,
            overlap_ratio=overlap_ratio,
        )

        chunks: list[dict[str, Any]] = []
        for level in levels:
            items = hier.get(level) or []
            # гарантируем формат dict с id/text/metadata
            for ch in items:
                if not isinstance(ch, dict):
                    continue
                text = (ch.get("text") or "").strip()
                if not text:
                    continue
                chunks.append(
                    {
                        "id": ch.get("id"),
                        "text": text,
                        "metadata": ch.get("metadata") or {},
                    }
                )

        return chunks

# src/engine/content/chunking.py

"""
Утилиты для работы с текстовыми чанками.

Здесь нет работы с БД и MinIO. Только чистые функции:
- построение summary для чанка;
- разбиение текста клаузы на под-чанки (по предложениям и абзацам);
- построение "section"-чанков из списка клауз.
"""

from __future__ import annotations

from typing import Any, Dict, List

from engine.retrieval.tokenization import count_tokens


def build_chunk_summary(path_titles: List[str], text: str, max_len: int = 200) -> str:
    """
    Построить короткое текстовое summary для чанка.

    Для непрофессионального программиста:
    - summary — это короткое описание чанка.
    - Мы склеиваем путь заголовков (path_titles) и первые 1–2 предложения текста.
    - Это нужно для:
      * более понятного UI,
      * более качественного поиска (retrieval/rerank по summary).
    """
    import re

    path = " / ".join(path_titles) if path_titles else ""
    text = (text or "").strip()
    if not text:
        return path

    # Берём первые 1–2 предложения.
    parts = re.split(r"(?<=[.!?])\s+", text)
    head = " ".join(parts[:2]).strip()

    if path and head:
        base = f"{path}: {head}"
    else:
        base = head or path

    if len(base) > max_len:
        return base[:max_len]
    return base


def split_text_to_sized_chunks(
    text: str,
    target_tokens: int,
    max_tokens: int,
    overlap_ratio: float,
) -> List[str]:
    """
    Режет текст клаузы на несколько меньших текстов (под-чанков).

    Логика:
    1) Если текст и так короткий (меньше max_tokens) и не выглядит как "простыня" —
       возвращаем его как есть.
    2) Иначе:
       - делим на абзацы по двойному переводу строки ("\n\n");
       - внутри абзацев делим на предложения по простому правилу:
         "точка/вопросительный/восклицательный знак + пробел + заглавная буква";
       - собираем чанки подряд из предложений, пока не наберём target_tokens,
         но не превысим max_tokens;
       - между чанками переносим "хвост" предложений (overlap), чтобы LLM
         не теряла контекст на границе чанков.
    """
    import re

    text = (text or "").strip()
    if not text:
        return []

    # Дополнительный предохранитель по длине строки:
    # если текст КРАЙНЕ длинный, не доверяем одному чанку даже если "по словам"
    # он проходит лимит, и всё равно пойдём в разбиение.
    TOO_LONG_CHARS = 10_000
    is_too_long = len(text) > TOO_LONG_CHARS

    # Если весь текст клаузы и так не больше max_tokens и по символам
    # он не выглядит как "простыня", можно не резать.
    if count_tokens(text) <= max_tokens and not is_too_long:
        return [text]

    # 1. Делим текст на абзацы.
    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]

    # 2. Из каждого абзаца извлекаем предложения.
    sentences: List[str] = []
    for p in paragraphs:
        # Очень простой split по предложениям.
        parts = re.split(r"(?<=[.!?])\s+(?=[A-ZА-Я0-9])", p)
        for s in parts:
            s = s.strip()
            if s:
                sentences.append(s)

    # Если по какой-то причине не смогли выделить предложения — вернём исходный текст.
    if not sentences:
        return [text]

    chunks: List[str] = []
    current_sentences: List[str] = []
    current_tokens = 0

    # "Хвост" предыдущего чанка для overlap.
    overlap_window: List[str] = []

    def _tokens_of(parts: List[str]) -> int:
        return count_tokens(" ".join(parts))

    for sent in sentences:
        sent_tokens = count_tokens(sent)

        # Если сейчас нет активного чанка, но есть overlap — начинаем с overlap.
        if not current_sentences and overlap_window:
            current_sentences.extend(overlap_window)
            current_tokens = _tokens_of(current_sentences)

        # Если предложение само по себе очень длинное (больше max_tokens) и
        # мы сейчас не собираем другой чанк — кладём его отдельным чанком.
        if sent_tokens > max_tokens and not current_sentences:
            chunks.append(sent)
            overlap_window = []
            current_sentences = []
            current_tokens = 0
            continue

        # Если добавление предложения переполнит max_tokens — закрываем текущий чанк.
        if current_sentences and current_tokens + sent_tokens > max_tokens:
            chunks.append(" ".join(current_sentences).strip())
            overlap_window = take_overlap_window(
                current_sentences,
                overlap_ratio=overlap_ratio,
            )
            current_sentences = []
            current_tokens = 0

        # Добавляем предложение в текущий чанк.
        current_sentences.append(sent)
        current_tokens += sent_tokens

        # Если мы уже достигли целевого размера — тоже закрываем чанк.
        if current_tokens >= target_tokens:
            chunks.append(" ".join(current_sentences).strip())
            overlap_window = take_overlap_window(
                current_sentences,
                overlap_ratio=overlap_ratio,
            )
            current_sentences = []
            current_tokens = 0

    # Если после цикла остались предложения — добавляем их последним чанком.
    if current_sentences:
        chunks.append(" ".join(current_sentences).strip())

    # Убираем пустые строки, если вдруг образовались.
    result = [c for c in chunks if c.strip()]

    return result or [text]


def take_overlap_window(
    sentences: List[str],
    overlap_ratio: float,
) -> List[str]:
    """
    Взять "хвост" из последних предложений для overlap между чанками.

    Для непрофессионального программиста:
    - overlap нужен, чтобы кусочек конца предыдущего чанка попал
      в начало следующего. Тогда модель не потеряет контекст на разрыве.
    - overlap_ratio = 0.15 означает "взять примерно 15% последних предложений".
    """
    if not sentences or overlap_ratio <= 0:
        return []

    n = max(1, int(len(sentences) * overlap_ratio))
    return sentences[-n:]


def build_section_chunks_from_clauses(
    doc_id: str,
    clause_chunks: List[Dict[str, Any]],
    max_tokens: int = 1200,
) -> List[Dict[str, Any]]:
    """
    Собрать "крупные" section-чанки на основе клауз.

    Для непрофессионального программиста:
    - Клаузы уже содержат путь заголовков (path_titles).
    - Мы группируем клаузы по первым 1–2 заголовкам пути
      и склеиваем их текст в один крупный блок.
    - Это даёт более высокий уровень "разделов", пригодный для
      обзора документа и контекстного retrieval.
    """
    from collections import defaultdict

    # Группируем клаузы по "шапке" пути заголовков (первые 2 уровня).
    groups: dict[tuple[str, ...], list[Dict[str, Any]]] = defaultdict(list)

    for ch in clause_chunks:
        meta = ch.get("metadata") or {}
        path = meta.get("path_titles") or []
        # Берём первые два уровня заголовков как ключ раздела.
        key = tuple(path[:2]) if path else ()
        if not key:
            # Клаузы без явного пути заголовков можно сгруппировать
            # отдельно, но сейчас их просто пропускаем.
            continue
        groups[key].append(ch)

    sections: List[Dict[str, Any]] = []

    for path_key, items in groups.items():
        if not items:
            continue

        # Сортируем клаузы внутри раздела по страницам и UID (если есть),
        # чтобы текст был в правильном порядке.
        def _sort_key(c: Dict[str, Any]) -> tuple:
            meta = c.get("metadata") or {}
            start_page = meta.get("start_page") or 0
            clause_uid = meta.get("clause_uid") or ""
            return (int(start_page), str(clause_uid))

        items_sorted = sorted(items, key=_sort_key)

        texts: List[str] = []
        pages: List[int] = []

        for it in items_sorted:
            t = (it.get("text") or "").strip()
            if t:
                texts.append(t)
            meta = it.get("metadata") or {}
            sp = meta.get("start_page")
            ep = meta.get("end_page")
            if isinstance(sp, int):
                pages.append(sp)
            if isinstance(ep, int):
                pages.append(ep)

        full_text = "\n\n".join(texts).strip()
        if not full_text:
            continue

        tokens = count_tokens(full_text)
        # Если раздел слишком длинный, можно будет в будущем порезать его
        # по абзацам, но сейчас оставим как есть.
        if tokens > max_tokens:
            # На будущее: можно вызвать split_text_to_sized_chunks
            # для full_text и сделать несколько section-под-чанков.
            pass

        section_id = f"{doc_id}:section:{' / '.join(path_key)}"
        start_page = min(pages) if pages else None
        end_page = max(pages) if pages else None
        path_titles = list(path_key)

        metadata = {
            "doc_id": str(doc_id),
            "clause_uid": None,
            "title": path_titles[-1] if path_titles else None,
            "level": 1,
            "path_titles": path_titles,
            "start_page": start_page,
            "end_page": end_page,
            "element_ids": [],
            "token_count": tokens,
            "summary": build_chunk_summary(path_titles, full_text),
        }

        # Аккуратно подтягиваем важные служебные поля из клауз в раздел:
        # - language_filter (target_language, degraded и пр.);
        # - position_ratio (если есть у клауз).
        lf_merged: dict[str, Any] = {}
        positions: list[float] = []

        for it in items_sorted:
            meta_it = it.get("metadata") or {}
            lf_it = meta_it.get("language_filter") or {}
            if lf_it:
                # Не переопределяем уже проставленные поля, чтобы сохранять
                # первый встретившийся "диагностический" контекст.
                for k, v in lf_it.items():
                    lf_merged.setdefault(k, v)

            pos_it = meta_it.get("position_ratio")
            if isinstance(pos_it, (int, float)):
                positions.append(float(pos_it))

        if lf_merged:
            metadata["language_filter"] = lf_merged
        if positions:
            # Для раздела логичнее взять медиану/среднее по клаузам,
            # здесь возьмём простое среднее.
            metadata["position_ratio"] = sum(positions) / len(positions)

        sections.append(
            {
                "id": section_id,
                "text": full_text,
                "metadata": metadata,
            }
        )

    return sections

# src/engine/content/chunk_exporter.py

"""
Экспорт документа в виде последовательности чанков в Markdown.

Модуль берёт нормализованные чанки (sized chunks) по документу,
склеивает их в один .md‑файл с понятными заголовками и путями
по структуре документа и сохраняет файл в MinIO.
Этот файл используется в debug-инструментах и для ручной проверки,
как система разрезала документ на части.
"""

from __future__ import annotations

import uuid
from typing import List

from sqlalchemy.ext.asyncio import AsyncSession

from engine.content.document_content_service import DocumentContentService
from storage.artifact_store import ArtifactStore, ArtifactPath
from db.repositories.document_repo import DocumentRepository


class ChunkExportService:
    """
    Сервис для выгрузки документа в виде последовательности чанков.

    Используется в debug-инструментах и для внутренних проверок:
    собирает все sized-чанки документа в один Markdown-файл и
    сохраняет его в MinIO, возвращая S3-URI.
    """

    def __init__(self, db: AsyncSession, store: ArtifactStore | None = None) -> None:
        self.db = db
        self.store = store or ArtifactStore()
        self.doc_repo = DocumentRepository(db)

    async def export_sized_chunks_markdown(
        self,
        *,
        doc_id: uuid.UUID,
        deal_id: uuid.UUID,
        tenant_id: uuid.UUID,
    ) -> str:
        """
        Собрать все sized-чанки документа в один Markdown-файл и сохранить в MinIO.

        Для непрофессионального пользователя:
        - берём уже посчитанные чанки нормального размера (get_sized_chunks);
        - для каждого чанка добавляем «шапку» с номером, пунктом договора,
          примерным размером и диапазоном страниц;
        - ниже выводим сам текст чанка;
        - в конце каждого чанка ставим разделитель `---`.

        Это нужно, чтобы юрист или аналитик мог глазами увидеть,
        как система разрезала документ на части, и проверить,
        не потерялся ли смысл на стыках чанков.

        Важно:
        - сюда уже должны прийти корректные tenant_id и deal_id,
          обычно их берут из объекта Deal (сделка по документу).

        Возвращается строка вида s3://... — ссылка на файл в хранилище.
        """
        # Для непрофессионального программиста:
        # - DocumentContentService умеет доставать "нормализованные чанки" по документу,
        # - здесь мы просто просим у него готовый список чанков.
        content = DocumentContentService(session=self.db)
        chunks = await content.get_sized_chunks(doc_id)

        if not chunks or not any((ch.get("text") or "").strip() for ch in chunks):
            raise ValueError(f"No sized chunks for document {doc_id}")        

        # Проверяем, что документ существует (на случай расхождений в БД).
        doc = await self.doc_repo.get_by_id(doc_id)
        if doc is None:
            raise ValueError(f"Document {doc_id} not found")

        # Строим человекочитаемый Markdown.
        lines: List[str] = []
        lines.append(f"# Document {doc_id}")
        lines.append("")

        for idx, ch in enumerate(chunks):
            meta = ch.get("metadata") or {}
            clause_uid = meta.get("clause_uid")
            tokens = meta.get("token_count")
            start_page = meta.get("start_page")
            end_page = meta.get("end_page")
            path = meta.get("path_titles") or []

            lines.append(
                f"## CHUNK {idx} "
                f"(clause {clause_uid or '-'}, tokens={tokens}, "
                f"pages {start_page}–{end_page})"
            )
            if path:
                lines.append(f"PATH: {' / '.join(path)}")

            lines.append("")
            lines.append(ch.get("text") or "")
            lines.append("")
            lines.append("---")
            lines.append("")

        md_text = "\n".join(lines)
        data = md_text.encode("utf-8")

        # Для непрофессионального программиста:
        # - ArtifactPath.export строит "правильный" путь в MinIO:
        #   tenants/{tenant_id}/deals/{deal_id}/exports/{doc_id}/{doc_id}__chunks_view.md
        # - tenant_id и deal_id мы уже получили снаружи (из Deal).
        export_path = ArtifactPath.export(
            tenant_id=str(tenant_id),
            deal_id=str(deal_id),
            export_id=str(doc_id),
            filename=f"{doc_id}__chunks_view.md",
        )
        uri = self.store.put_file(export_path, data, content_type="text/markdown")
        return uri

"""
Clause tree → клауза-чанки.

Здесь:
- вся логика обхода clause tree;
- сборка текста клаузы по element_ids из layout;
- универсальные хелперы для:
  * root-узлов,
  * индекса элементов layout.
"""

from __future__ import annotations

import uuid
from typing import Any, Dict, List


class ClauseTreeBuilder:
    """
    Класс-хелпер для работы с clause tree.

    Для непрофессионального программиста:
    - Этот класс не знает про БД и MinIO.
    - Ему дают уже загруженный JSON clause tree и JSON layout,
      и он возвращает список "чистых" чанков по листовым клаузам.
    """

    @staticmethod
    def extract_root_children(tree: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Извлечь список узлов верхнего уровня из clause tree JSON.

        Формат ClauseTree.to_dict():
          {
            "document_id": "...",
            "root": [ {...}, {...} ],
            ...
          }

        На всякий случай поддерживаем и формат:
          { "root": { "children": [ {...}, ... ] } }
        """
        raw_root = tree.get("root")

        if isinstance(raw_root, list):
            return raw_root

        if isinstance(raw_root, dict):
            children = raw_root.get("children")
            if isinstance(children, list):
                return children

        return []

    @staticmethod
    def build_element_index(layout: Dict[str, Any]) -> Dict[str, Dict[str, Any]]:
        """
        Построить индекс element_id → элемент layout'а.

        Это ускоряет сбор текста клаузы: вместо линейного поиска по всему
        списку elements мы обращаемся к нужному элементу напрямую по ключу.

        Формат элемента (см. build_text_layout_raw):
          {
            "element_id": str,
            "type": "heading" | "text" | "listitem",
            "text": str,
            "page": int,
            ...
          }
        """
        elements = layout.get("elements") or []
        index: Dict[str, Dict[str, Any]] = {}
        for el in elements:
            el_id = el.get("element_id")
            if el_id:
                index[el_id] = el
        return index

    @classmethod
    def build_leaf_chunks(
        cls,
        doc_id: uuid.UUID,
        tree: Dict[str, Any],
        layout: Dict[str, Any],
    ) -> List[Dict[str, Any]]:
        """
        Построить клауза-чанки (листовые узлы) на основе clause tree + layout.

        Возвращает:
        [
          {
            "id": "<doc_id>:<clause_uid>",
            "text": "<полный текст клаузы>",
            "metadata": {...}
          },
          ...
        ]
        """
        root_children = cls.extract_root_children(tree)
        if not root_children:
            return []

        element_index = cls.build_element_index(layout)

        chunks: List[Dict[str, Any]] = []
        for child in root_children:
            cls._collect_leaf_chunks(
                doc_id=doc_id,
                node=child,
                element_index=element_index,
                out=chunks,
                path_titles=[],
            )

        return chunks

    @classmethod
    def _collect_leaf_chunks(
        cls,
        doc_id: uuid.UUID,
        node: Dict[str, Any],
        element_index: Dict[str, Dict[str, Any]],
        out: List[Dict[str, Any]],
        path_titles: List[str] | None = None,
    ) -> None:
        """
        Рекурсивно обойти дерево и добавить чанки для всех листовых клауз
        (то есть узлов без children).

        Для каждой листовой клаузы:
        - берём все её element_ids,
        - находим соответствующие элементы в layout,
        - склеиваем их текст в одном чанке.
        """
        # path_titles — это "хлебные крошки" из заголовков от корня до текущего узла.
        current_path: List[str] = list(path_titles or [])
        node_title = node.get("title")
        if node_title:
            current_path.append(node_title)

        children = node.get("children") or []

        if not children:
            clause_uid = node.get("clause_uid")
            element_ids: List[str] = node.get("element_ids") or []

            # Собираем текст клаузы из элементов layout.
            texts: List[str] = []
            pages: List[int] = []
            last_page: int | None = None

            for el_id in element_ids:
                el = element_index.get(el_id)
                if not el:
                    continue

                raw_text = el.get("text") or ""
                text = raw_text.strip()
                if not text:
                    continue

                page = el.get("page")
                if isinstance(page, int):
                    pages.append(page)

                # Если элемент на новой странице — добавляем пустую строку
                # как явный разделитель, чтобы модели видели разрыв страниц.
                if last_page is not None and page is not None and page != last_page:
                    texts.append("")

                texts.append(text)
                last_page = page

            # Склеиваем блоки через двойной перевод строки, чтобы сохранить структуру.
            # Пустые строки, добавленные при смене страницы, превратятся в лишние
            # пустые абзацы, что удобно для LLM.
            full_text = "\n\n".join(texts).strip()

            if not full_text:
                # Для непрофессионального программиста:
                # - если после фильтрации / санитизации в этой листовой клаузе
                #   не осталось ни одного непустого текстового элемента,
                #   пустой чанк создавать не нужно.
                return

            start_page = min(pages) if pages else node.get("start_page")
            end_page = max(pages) if pages else node.get("end_page")

            metadata = {
                "doc_id": str(doc_id),
                "clause_uid": clause_uid,
                "title": node.get("title"),
                "level": node.get("level"),
                "path_titles": current_path,
                "start_page": start_page,
                "end_page": end_page,
                "element_ids": element_ids,
            }

            out.append(
                {
                    "id": f"{doc_id}:{clause_uid}",
                    "text": full_text,
                    "metadata": metadata,
                }
            )
            return

        # Если у узла есть дети — спускаемся ниже по дереву.
        for child in children:
            cls._collect_leaf_chunks(
                doc_id=doc_id,
                node=child,
                element_index=element_index,
                out=out,
                path_titles=current_path,
            )

# src/api/endpoints/document_chunk_exports.py

"""
Эндпоинты для экспорта документа в виде «чанков» (chunks).

Для чего это нужно непрофессиональному пользователю (юристу / аналитику):

- Документ автоматически режется на небольшие логические куски — чанки.
- В каждом чанке система старается сохранить связный фрагмент текста
  (абзац, подпункт или небольшую группу пунктов).
- Этот модуль позволяет получить специальный .md‑файл, где:
  * каждый чанк идёт один за другим;
  * перед текстом чанка есть заголовок с:
    - номером чанка,
    - идентификатором пункта договора (clause_uid), если он известен,
    - примерным размером чанка по токенам/словам,
    - диапазоном страниц в оригинальном PDF;
  * при наличии структуры (оглавление/путь разделов) она показывается
    перед чанком в виде PATH: Раздел / Подраздел / Пункт.

Файл нужен для ручной проверки качества чанкинга:
- юрист может глазами посмотреть, где система «режет» документ;
- увидеть, не попали ли важные абзацы на стык чанков;
- сравнить текст с исходным PDF, если нужно.

Важно: этот эндпоинт НИЧЕГО не меняет в базе данных и в документе.
Он только:
- забирает уже посчитанные чанки документа,
- собирает из них один большой Markdown‑текст,
- сохраняет его во внутреннем хранилище (MinIO / S3),
- возвращает HTTP‑ссылку для скачивания/просмотра.
"""

from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession

from api.deps.auth import CurrentUser, get_current_user
from api.deps.security import require_project_role

router = APIRouter(

@router.get(

async def get_document_chunk_view(
    document_id: UUID,
    db: AsyncSession = Depends(get_db),
    user: CurrentUser = Depends(get_current_user),
) -> dict:
    """
    Вернуть ссылку на специальный .md‑файл, где документ показан
    «как его видит система чанкинга».

    Для непрофессионального пользователя:

    - Документ режется на небольшие смысловые куски (чанки).
    - В результирующем файле каждый чанк идёт один за другим,
      с заголовком, где указаны:
        * номер чанка;
        * пункт договора (clause_uid), если он известен;
        * примерный размер по токенам;
        * диапазон страниц в оригинальном PDF.
    - Если у документа есть оглавление/структура, перед чанком
      показывается путь до него (PATH: Раздел / Подраздел / Пункт).

    Это помогает юристу:

    - увидеть, как система «разрезала» документ;
    - проверить, не потерялся ли смысл на стыках чанков;
    - сравнить разбиение с исходным PDF.

    Технически эндпоинт:

    - проверяет, что документ существует и принадлежит проекту пользователя;
    - проверяет роль пользователя в проекте (viewer и выше);
    - находит сделку документа и её tenant_id;
    - собирает Markdown‑представление чанков через ChunkExportService;
    - сохраняет .md‑файл во внутреннем хранилище (MinIO / S3);
    - конвертирует внутренний s3:// URI в обычную HTTP‑ссылку;
    - возвращает JSON-объект вида: { "url": "https://.../doc_id__chunks_view.md" }.

    Важно: эндпоинт не запускает заново обработку документа и не меняет его
    статус. Он использует уже готовые sized‑чанки.
    """

    # 1. Проверяем, что документ существует
    doc_repo = DocumentRepository(db)
    doc = await doc_repo.get_by_id(document_id)
    if doc is None:
        # Для пользователя: документа не существует или он был удалён.
        raise HTTPException(status_code=404, detail="Document not found")

    # 2. Проверяем, что пользователь имеет права на проект этого документа.
    #    Допускаем: admin, editor, reviewer, viewer.
    await require_project_role(
        db=db,
        user=user,
        deal_id=doc.deal_id,
        allowed_roles=("admin", "editor", "reviewer", "viewer"),
    )

    # 3. Находим сделку, чтобы получить tenant_id.
    #    Для непрофессионального программиста:
    #    - документ "живёт" внутри сделки (Deal),
    #    - у сделки есть tenant_id (арендатор в системе),
    #    - поэтому путь в MinIO строится по паре (tenant_id, deal_id).
    deal_repo = DealRepository(db)
    deal = await deal_repo.get_by_id(doc.deal_id)
    if deal is None:
        # Это аномальная ситуация: документ ссылается на несуществующую сделку.
        logger.error(
            "Deal not found for document when preparing chunk view",
            extra={
                "document_id": str(document_id),
                "deal_id": str(doc.deal_id),
            },
        )
        raise HTTPException(
            status_code=500,
            detail="Internal error: deal not found for this document.",
        )

    # 4. Собираем Markdown‑файл из уже посчитанных чанков.
    exporter = ChunkExportService(db)
    try:
        # s3_uri — внутренний путь вида s3://bucket/tenants/.../exports/...
        s3_uri = await exporter.export_sized_chunks_markdown(
            doc_id=document_id,
            deal_id=deal.id,
            tenant_id=deal.tenant_id,
        )
    except ValueError as e:
        # Например, если DocumentContentService не нашёл чанков
        # или вернул некорректные данные.
        logger.warning(
            "Chunk export failed with ValueError",
            extra={"document_id": str(document_id), "error": str(e)},
        )
        raise HTTPException(
            status_code=404,
            detail="Chunk view is not available for this document yet",
        )
    except Exception as e:
        # Любая другая непредвиденная ошибка — логируем и отдаём 500 с понятным detail.
        logger.exception(
            "Unexpected error during chunk export",
            extra={"document_id": str(document_id)},
        )
        raise HTTPException(
            status_code=500,
            detail="Internal error while preparing chunk view for this document.",
        ) from e

    # 5. Превращаем внутренний s3:// путь в ПОДПИСАННУЮ HTTP‑ссылку (presigned URL),
    #    по которой браузер сможет скачать или открыть файл даже при приватном бакете.
    store = ArtifactStore()
    presigned_url = store.get_presigned_get_url(s3_uri)

    # 6. Возвращаем JSON с единственным полем "url".
    #    На фронте этим URL можно:
    #    - либо открыть в новой вкладке,
    #    - либо скачать как файл.
    return {"url": presigned_url}

# src/services/structure/builder.py

"""
Structure Builder: parses text_layout_raw elements into a clause tree.

Supports:
  - Numeric: 1., 1.1, 1.1.1, 1.1 (без точки после номера)
  - Roman: I, II, III, IV (level 1 only)
  - Alphabetic: A., B., a), b) (level 2)
  - Cross-reference detection: см. п. X.X, Art. X
  - TOC detection: автоматическое подавление блока оглавления
  - Degraded mode: flat chunks when coverage < 60%
"""

import re

from core.config import settings
from core.logging import get_logger
from services.structure.models import ClauseNode, ClauseTree, CrossReference

logger = get_logger(__name__)

# Numeric: "1.", "1.1.", "1.1.1." — с точкой после номера (строгий)
_NUMERIC_FULL = re.compile(r"^(\d+(?:\.\d+)*)[\.\)]\s+(.{1,120})")

# Numeric без точки: "1.1 Title", "2.3.1 Title" — только для вложенных (есть хотя бы одна точка)
# Требуем заглавную букву или «» после пробела — защита от числовых данных
_NUMERIC_NO_DOT = re.compile(
    r"^(\d+(?:\.\d+)+)\s{1,3}([A-ZА-ЯЁ\«\"].{1,120})"
)

_ROMAN_L1 = re.compile(
    r"^(I{1,3}|IV|V?I{0,3}|IX|X{0,3}(?:IX|IV|V?I{0,3}))[\.\)]\s+(.{1,120})",
    re.IGNORECASE,
)
_ALPHA = re.compile(
    r"^([A-Fa-f\u0430-\u0436])[\.)\s]\s+(.{1,120})",
)
_CROSS_REF = re.compile(
    r"(?:см\.\s*п\.?|ст\.\s*|пункт\s+|п\.?\s*)(\d+(?:\.\d+)*)|"
    r"(?:Art(?:icle)?\.?\s*)(\d+(?:\.\d+)*)",
    re.IGNORECASE,
)

# Маркеры начала блока оглавления (любой регистр)
_TOC_MARKERS = frozenset([
    "table of contents",
    "contents",
    "содержание",
    "оглавление",
])

# Максимальное число страниц, на которых может располагаться оглавление.
# Оглавление практически никогда не занимает больше 5 страниц.
_TOC_MAX_PAGES = 5


def _level_from_uid(uid: str) -> int:
    return uid.count(".") + 1


def _parse_element_as_heading(text: str) -> tuple | None:
    stripped = text.strip()
    # Порядок важен: строгий numeric (с точкой) проверяем первым,
    # затем numeric без точки (только вложенные), затем roman и alpha.
    for pattern in (_NUMERIC_FULL, _NUMERIC_NO_DOT, _ROMAN_L1, _ALPHA):
        m = pattern.match(stripped)
        if m:
            uid, title = m.group(1), m.group(2).strip()
            return uid, title[: settings.STRUCTURE_MAX_CLAUSE_TITLE_LEN]

    # Fallback для случаев, когда после language‑filter заголовок чуть "обрезался",
    # и не попал под строгие паттерны, но очевидно начинается с числового uid.
    # Примеры: "1 Definitions", "2.3 Other terms" без точки и с коротким title.
    m_simple = re.match(r"^(\d+(?:\.\d+)*)(?:\s+(.+))?$", stripped)
    if m_simple:
        uid = m_simple.group(1)
        title = (m_simple.group(2) or "").strip()
        if not title:
            # Если title совсем пустой, лучше не считать это заголовком.
            return None
        return uid, title[: settings.STRUCTURE_MAX_CLAUSE_TITLE_LEN]

    return None


class StructureBuilder:
    """Builds a clause tree from text_layout_raw elements."""

    def build(
        self,
        document_id: str,
        elements: list[dict],
    ) -> ClauseTree:

        # Попробуем понять, был ли layout помечен как degraded на стадии language filter.
        # Metadata обычно лежит на уровне layout, но до нас она доходит только
        # через элементы, поэтому смотрим на первый попадшийся language_filter.
        layout_degraded = False
        layout_degraded_reason: str | None = None
        for el in elements:
            lf = (el.get("language_filter") or {}) if isinstance(el, dict) else {}
            degraded_flag = lf.get("degraded")
            if degraded_flag is not None:
                layout_degraded = bool(degraded_flag)
                layout_degraded_reason = lf.get("degraded_reason")
                break


        # Определяем конец блока оглавления и работаем только с телом документа.
        # Это защищает от ситуации, когда строки оглавления ("1. Определения",
        # "2. Предмет договора") парсятся как реальные заголовки и создают
        # пустые чанки вместо содержательных.
        toc_end_idx = self._find_toc_end(elements)
        body_elements = elements[toc_end_idx:]

        if not body_elements:
            # На случай если весь документ был ошибочно принят за оглавление
            body_elements = elements
            toc_end_idx = 0

        headings: list[tuple] = []
        for i, el in enumerate(body_elements):
            text = el.get("text", "")
            parsed = _parse_element_as_heading(text)
            if parsed:
                uid, title = parsed
                headings.append((i, uid, title, el.get("page", 1)))

        if not headings:
            return self._degraded_tree(document_id, body_elements)

        coverage = self._calculate_coverage(headings, body_elements)
        confidence, degraded = self._confidence_and_mode(coverage)

        # Если уже на стадии language filter layout был помечен как degraded,
        # а coverage здесь не выглядит идеально, усиливаем решение в сторону degraded.
        if layout_degraded and not degraded and coverage < 0.8:
            logger.info(
                "Structure: forcing degraded mode due to degraded primary layout",
                document_id=document_id,
                coverage=round(coverage, 3),
                layout_degraded_reason=layout_degraded_reason,
            )
            degraded = True
            confidence = "LOW"

        if degraded:
            logger.info(
                "Structure degraded: low coverage",
                document_id=document_id,
                coverage=round(coverage, 3),
                toc_skipped_elements=toc_end_idx,
            )
            return self._degraded_tree(document_id, body_elements, coverage)

        root_nodes, references = self._build_tree(headings, body_elements)
        return ClauseTree(
            document_id=document_id,
            root=root_nodes,
            references_index=references,
            structure_confidence=confidence,
            degraded=False,
        )

    def _find_toc_end(self, elements: list[dict]) -> int:
        """
        Найти индекс первого элемента ПОСЛЕ блока оглавления.

        Стратегия:
        1) Ищем явный маркер оглавления ("TABLE OF CONTENTS", "СОДЕРЖАНИЕ" и т.п.).
           Если нашли — ищем конец блока оглавления после него.
        2) Если явного маркера нет — ищем кластер: несколько подряд идущих элементов,
           все проходящих как заголовки с последовательными uid верхнего уровня
           ("1", "2", "3"...) без вложенных подразделов между ними.
           Такой кластер — признак оглавления.
        3) Если ничего не нашли — возвращаем 0 (оглавления нет, обрабатываем всё).

        Ограничение: ищем только в первых _TOC_MAX_PAGES страницах.
        """
        if not elements:
            return 0

        # Определяем страницу-порог: не смотрим дальше _TOC_MAX_PAGES
        first_page = elements[0].get("page", 1)
        page_threshold = first_page + _TOC_MAX_PAGES

        # Шаг 1: поиск явного маркера
        toc_start_idx: int | None = None
        for i, el in enumerate(elements):
            if el.get("page", 1) > page_threshold:
                break
            text_lower = (el.get("text") or "").strip().lower()
            if text_lower in _TOC_MARKERS:
                toc_start_idx = i
                break

        if toc_start_idx is not None:
            # Нашли маркер — ищем конец блока оглавления.
            # Конец оглавления = первый элемент после маркера, который:
            # а) находится на новой странице относительно последней строки оглавления
            #    И при этом является заголовком (начало реального раздела),
            # б) ИЛИ не является заголовком и при этом достаточно длинный (>80 символов,
            #    явно не строка оглавления).
            # Дополнительное ограничение: не выходим за page_threshold.
            toc_last_page = elements[toc_start_idx].get("page", 1)

            for i in range(toc_start_idx + 1, len(elements)):
                el = elements[i]
                el_page = el.get("page", 1)

                if el_page > page_threshold:
                    # Вышли за лимит страниц — считаем что оглавление закончилось
                    return i

                text = (el.get("text") or "").strip()
                parsed = _parse_element_as_heading(text)

                # Если мы на новой странице относительно последней строки оглавления
                # и встретили заголовок — это начало реального тела документа
                if el_page > toc_last_page and parsed:
                    return i

                # Длинный не-заголовочный текст — явно тело документа
                if not parsed and len(text) > 80:
                    return i

                # Продолжаем сканировать оглавление
                toc_last_page = el_page

            # Прошли все элементы в пределах порога — оглавление не закончилось,
            # возвращаем индекс первого элемента за порогом страниц
            return toc_start_idx + 1  # минимальная защита

        # Шаг 2: поиск кластера без явного маркера.
        # Кластер оглавления: ≥3 подряд идущих элементов, все являются заголовками
        # верхнего уровня (uid без точек: "1", "2", "3"...), строго последовательные.
        cluster_start: int | None = None
        cluster_uids: list[str] = []
        cluster_last_idx: int = 0

        for i, el in enumerate(elements):
            if el.get("page", 1) > page_threshold:
                break
            text = (el.get("text") or "").strip()
            parsed = _parse_element_as_heading(text)

            if parsed:
                uid, _ = parsed
                # Только верхний уровень (без точки в uid) для детектирования кластера
                if "." not in uid:
                    if cluster_start is None:
                        cluster_start = i
                        cluster_uids = [uid]
                    else:
                        cluster_uids.append(uid)
                    cluster_last_idx = i
                    continue

            # Не-заголовочный элемент прерывает кластер
            if cluster_start is not None and len(cluster_uids) >= 3:
                # Нашли кластер из ≥3 верхнеуровневых заголовков подряд — это оглавление.
                # Возвращаем индекс следующего после кластера элемента.
                return cluster_last_idx + 1

            # Сбрасываем кластер
            cluster_start = None
            cluster_uids = []

        # Проверяем незакрытый кластер в конце диапазона
        if cluster_start is not None and len(cluster_uids) >= 3:
            return cluster_last_idx + 1

        return 0  # оглавление не обнаружено

    def _build_tree(
        self,
        headings: list,
        elements: list,
    ) -> tuple[list[ClauseNode], list[CrossReference]]:
        nodes: dict[str, ClauseNode] = {}
        root: list[ClauseNode] = []

        for h_idx, (elem_i, uid, title, page) in enumerate(headings):
            next_elem_i = (
                headings[h_idx + 1][0] if h_idx + 1 < len(headings) else len(elements)
            )
            element_ids = [
                elements[j].get("element_id", f"e{j:04d}")
                for j in range(elem_i, next_elem_i)
            ]
            end_page = (
                elements[next_elem_i - 1].get("page", page)
                if next_elem_i > elem_i
                else page
            )

            node = ClauseNode(
                clause_uid=uid,
                title=title,
                level=_level_from_uid(uid),
                element_ids=element_ids,
                start_page=page,
                end_page=end_page,
            )

            # Если uid уже существует в nodes — это дубль (например, два "1" в документе
            # после некорректного парсинга). Сохраняем только первый встреченный,
            # последующие дубли добавляем в root как отдельные узлы без регистрации
            # в nodes, чтобы они не ломали привязку дочерних узлов.
            if uid not in nodes:
                nodes[uid] = node

            parent_uid = self._find_parent_uid(uid, nodes)
            if parent_uid and parent_uid in nodes:
                nodes[parent_uid].children.append(node)
            else:
                root.append(node)

        references = self._extract_references(elements, nodes)
        return root, references

    # --- Остальные методы без изменений ---

    def _find_parent_uid(self, uid: str, nodes: dict) -> str | None:
        parts = uid.rsplit(".", 1)
        if len(parts) > 1:
            parent = parts[0]
            return parent if parent in nodes else None
        return None

    def _calculate_coverage(self, headings: list, elements: list) -> float:
        if not elements:
            return 0.0
        assigned = set()
        for h_idx, (elem_i, uid, title, page) in enumerate(headings):
            next_elem_i = (
                headings[h_idx + 1][0] if h_idx + 1 < len(headings) else len(elements)
            )
            for j in range(elem_i, next_elem_i):
                assigned.add(j)
        return len(assigned) / len(elements)

    def _confidence_and_mode(self, coverage: float) -> tuple[str, bool]:
        threshold = settings.STRUCTURE_MIN_CLAUSE_COVERAGE
        if coverage < threshold:
            return "LOW", True
        elif coverage < 0.80:
            return "MEDIUM", False
        return "HIGH", False

    def _extract_references(
        self, elements: list, nodes: dict
    ) -> list[CrossReference]:
        references: list[CrossReference] = []
        uid_set = set(nodes.keys())
        for el in elements:
            text = el.get("text", "")
            el_id = el.get("element_id", "")
            source_uid = self._element_to_clause(el_id, nodes)
            if not source_uid:
                continue
            for match in _CROSS_REF.finditer(text):
                target = match.group(1) or match.group(2)
                if target and target in uid_set and target != source_uid:
                    references.append(
                        CrossReference(
                            source_clause_uid=source_uid,
                            target_clause_uid=target,
                            reference_text=match.group(),
                        )
                    )
        return references

    def _element_to_clause(self, element_id: str, nodes: dict) -> str | None:
        for uid, node in nodes.items():
            if element_id in node.element_ids:
                return uid
        return None

    def _degraded_tree(
        self,
        document_id: str,
        elements: list,
        coverage: float = 0.0,
    ) -> ClauseTree:
        chunks: list[ClauseNode] = []
        current_ids: list[str] = []
        current_tokens = 0
        chunk_idx = 1
        max_tokens = settings.STRUCTURE_DEGRADED_CHUNK_TOKENS
        current_start_page: int | None = None
        last_page: int = elements[0].get("page", 1) if elements else 1

        for el in elements:
            text = el.get("text", "")
            tokens = len(text.split())
            page = el.get("page", 1)

            if current_tokens + tokens > max_tokens and current_ids:
                start_page = current_start_page or last_page
                end_page = last_page
                chunks.append(
                    ClauseNode(
                        clause_uid=str(chunk_idx),
                        title=f"Chunk {chunk_idx}",
                        level=1,
                        element_ids=current_ids,
                        start_page=start_page,
                        end_page=end_page,
                    )
                )
                chunk_idx += 1
                current_ids = []
                current_tokens = 0
                current_start_page = None

            current_ids.append(el.get("element_id", f"e{chunk_idx:04d}"))
            current_tokens += tokens
            if current_start_page is None:
                current_start_page = page
            last_page = page

        if current_ids:
            chunks.append(
                ClauseNode(
                    clause_uid=str(chunk_idx),
                    title=f"Chunk {chunk_idx}",
                    level=1,
                    element_ids=current_ids,
                    start_page=current_start_page or last_page,
                    end_page=last_page,
                )
            )

        return ClauseTree(
            document_id=document_id,
            root=chunks,
            references_index=[],
            structure_confidence="LOW",
            degraded=True,
        )

# src/services/structure/models.py

"""
Clause tree data models.
"""

from dataclasses import dataclass, field


@dataclass
class ClauseNode:
    clause_uid: str  # "1", "1.1", "1.1.1", "II", "A"
    title: str  # truncated to MAX_CLAUSE_TITLE_LEN
    level: int  # 1 — top level
    element_ids: list = field(default_factory=list)
    start_page: int = 1
    end_page: int = 1
    children: list = field(default_factory=list)


@dataclass
class CrossReference:
    source_clause_uid: str
    target_clause_uid: str
    reference_text: str  # "см. п. 3.2"


@dataclass
class ClauseTree:
    document_id: str
    root: list  # top-level ClauseNode list
    references_index: list  # list[CrossReference]
    structure_confidence: str  # LOW | MEDIUM | HIGH
    degraded: bool = False

    def to_dict(self) -> dict:
        return {
            "document_id": self.document_id,
            "root": [self._node_to_dict(n) for n in self.root],
            "references_index": [
                {
                    "source": r.source_clause_uid,
                    "target": r.target_clause_uid,
                    "text": r.reference_text,
                }
                for r in self.references_index
            ],
            "structure_confidence": self.structure_confidence,
            "degraded": self.degraded,
        }

    def _node_to_dict(self, node: ClauseNode) -> dict:
        return {
            "clause_uid": node.clause_uid,
            "title": node.title,
            "level": node.level,
            "element_ids": node.element_ids,
            "start_page": node.start_page,
            "end_page": node.end_page,
            "children": [self._node_to_dict(c) for c in node.children],
        }
