const POLL_INTERVAL = 4000;

// ─────────────────────────────────────────────────────────────
// Deal wrapper
// ─────────────────────────────────────────────────────────────

function DealRouteWrapper() {
  const { dealId } = useParams<{ dealId: string }>();
  const { t } = useTranslation("common");
  const {
    data: deal,
    isLoading,
    isError,
    error,
  } = useDeal(dealId ?? null, {
    refetchInterval: (query) =>
      shouldPollDealLanguageDiscovery(
        query.state.data?.language_discovery_status,
      )
        ? POLL_INTERVAL
        : false,
  });

  if (isLoading) return <div>{t("loading")}</div>;

  if (isError || !deal) {
    return (
      <div>
        {error instanceof Error ? error.message : t("projectLoadError")}
      </div>
    );
  }

  return <DealPage deal={deal} />;
}

function DocTypeVersionLegacyRedirect() {
  const { docType, versionId } = useParams<{
    docType: string;
    versionId: string;
  }>();

  if (!docType) {
    return <Navigate to="/admin/instructions/doc-types" replace />;
  }

  // Старый /versions/new больше не используется как отдельная страница.
  // Безопасно отправляем пользователя на объединённый detail screen.
  if (!versionId || versionId === "new") {
    return <Navigate to={`/admin/instructions/doc-types/${docType}`} replace />;
  }

  return (
    <Navigate
      to={`/admin/instructions/doc-types/${docType}?version=${encodeURIComponent(versionId)}`}
      replace
    />
  );
}

// ─────────────────────────────────────────────────────────────
// Route Guards
// ─────────────────────────────────────────────────────────────

function RequireAuth({ children }: { children: React.ReactElement }) {
  const { userId } = useAuth();
  if (!userId) return <Navigate to="/login" replace />;
  return children;
}

function RequireSuperadmin({ children }: { children: React.ReactElement }) {
  const { userId, globalRole } = useAuth();
  if (!userId) return <Navigate to="/login" replace />;
  if (globalRole !== "superadmin") return <Navigate to="/dashboard" replace />;
  return children;
}

function RequireAdminOrSuperadmin({
  children,
}: {
  children: React.ReactElement;
}) {
  const { userId, globalRole } = useAuth();
  if (!userId) return <Navigate to="/login" replace />;
  if (globalRole !== "superadmin" && globalRole !== "tenant_admin") {
    return <Navigate to="/dashboard" replace />;
  }
  return children;
}

import React from "react";

/**
 * PathCrumb — editorial-style path-крошка в стиле Legora (PRODUCT / Tabular Review).
 *
 * Композиция:
 *   • root часть (например, «ПЛАТФОРМА») — приглушённая
 *   • слэш — ещё более приглушённый
 *   • leaf часть (например, «Анализ документов») — основной цвет
 *
 * Типографика — uppercase, tracking-широкий, малый размер (11px).
 * Используется как замена простому eyebrow на product-страницах.
 *
 * Пример:
 *   <PathCrumb root="ПЛАТФОРМА" leaf="Анализ документов" />
 */

interface PathCrumbProps {
  root: string;
  leaf: string;
  /**
   * Вариант цвета для тёмных секций (белый текст).
   * По умолчанию — светлая тема.
   */
  variant?: "light" | "dark";
}

const PathCrumb: React.FC<PathCrumbProps> = ({ root, leaf, variant = "light" }) => {
  const rootColor = variant === "dark" ? "rgba(250, 248, 245, 0.45)" : "#A8A29E";
  const slashColor = variant === "dark" ? "rgba(250, 248, 245, 0.28)" : "#D6D0C7";

  return (
    <div
      className="inline-flex items-baseline gap-2 text-[11px] font-medium uppercase tracking-[0.22em]"
      aria-label={`${root} / ${leaf}`}
    >
      <span style={{ color: rootColor }}>{root}</span>
      <span style={{ color: slashColor }} aria-hidden="true">
        /
      </span>
